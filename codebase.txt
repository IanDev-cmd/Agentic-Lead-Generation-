/**
 * Apps Script hub for the lead-generation webhook.
 *
 * Deploy as a web app that executes as you, accessible to anyone.
 * The shared WEBHOOK_SECRET is the access control, because Apps Script
 * anonymous web apps do not provide Google-authenticated callers for a local Node process.
 *
 * Run install() from the editor once and approve scopes before the first POST.
 * Drafts are created with GmailApp.createDraft. MailApp only sends mail, so it is not used.
 * Geocoding uses the built-in Maps service, which does not need a Maps Platform billing account.
 * Follow-up tasks use the Advanced Tasks service (Tasks API v1).
 *
 * Script properties:
 *   WEBHOOK_SECRET, SPREADSHEET_ID, BRIEFINGS_FOLDER_ID,
 *   SELLER_NAME, SELLER_OFFERING, HOME_BASE_ADDRESS,
 *   CALENDAR_LOOKAHEAD_DAYS, MAX_LEADS, TASKLIST_ID
 */

var LEAD_HEADERS = [
  'Generated At',
  'Interaction ID',
  'Company',
  'Domain',
  'HQ Address',
  'Latitude',
  'Longitude',
  'Distance Km',
  'Contact Name',
  'Contact Title',
  'Contact Email',
  'Pain Point',
  'Hook',
  'Priority',
  'Source',
  'Briefing URL',
  'Draft ID',
  'Calendar Event IDs',
  'Task ID',
  'Task Due',
  'Notes'
];

var RUN_HEADERS = ['Interaction ID', 'Processed At', 'Lead Count', 'Report URL', 'Status'];
var HARD_LEAD_CAP = 15;

function doGet(e) {
  var props = PropertiesService.getScriptProperties();
  var expected = props.getProperty('WEBHOOK_SECRET') || '';
  if (!expected || !secretsMatch_(providedSecret_(e), expected.trim())) {
    return json_({ ok: false, error: 'Unauthorized webhook. Render WEBHOOK_SECRET does not match this script property.' });
  }
  return json_({
    ok: true,
    service: 'lead-generation-webhook',
    spreadsheetConfigured: Boolean(props.getProperty('SPREADSHEET_ID')),
    folderConfigured: Boolean(props.getProperty('BRIEFINGS_FOLDER_ID'))
  });
}

function doPost(e) {
  try {
    return handlePost_(e);
  } catch (err) {
    Logger.log(err);
    return json_({
      ok: false,
      error: 'The webhook failed before the batch was saved.'
    });
  }
}

function install() {
  var props = PropertiesService.getScriptProperties();
  var spreadsheet = getOrCreateSpreadsheet_(props);
  var folder = getOrCreateFolder_(props);
  ensureSheet_(spreadsheet, 'Leads', LEAD_HEADERS);
  ensureSheet_(spreadsheet, 'Runs', RUN_HEADERS);
  Logger.log('SPREADSHEET_URL=' + spreadsheet.getUrl());
  Logger.log('BRIEFINGS_FOLDER_URL=' + folder.getUrl());
  if (!props.getProperty('WEBHOOK_SECRET')) {
    Logger.log('Set WEBHOOK_SECRET in Project Settings > Script properties before deploying.');
  }
  Logger.log('Change the project timezone if Etc/GMT-3 is not your timezone. Task due dates use it.');
}

function handlePost_(e) {
  var body = readBody_(e);
  var config = getConfig_(body);
  if (!config.secret) {
    return json_({ ok: false, error: 'WEBHOOK_SECRET is not set. Run install(), then set the script property.' });
  }
  if (!secretsMatch_(providedSecret_(e), config.secret)) {
    return json_({ ok: false, error: 'Unauthorized webhook. Render WEBHOOK_SECRET does not match this script property.' });
  }

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) {
    return json_({ ok: false, retryable: true, error: 'Script is busy. Retry this delivery.' });
  }

  try {
    if (String(body.action || '') === 'approve') {
      return approveLead_(body, config);
    }
    if (String(body.action || '') === 'links') return workspaceLinks_();
    if (String(body.action || '') === 'list') return listWorkspace_(body);
    if (String(body.action || '') === 'read') return readWorkspace_(body);

    var markdown = String(body.markdown || '');
  if (!markdown || markdown.length > 500000) {
    return json_({ ok: false, error: 'Markdown payload is missing or larger than 500000 characters.' });
  }
  var interactionId = String(body.interactionId || '').trim();
  if (!interactionId) {
    return json_({ ok: false, error: 'interactionId is required.' });
  }

  var props = PropertiesService.getScriptProperties();
    var spreadsheet = getOrCreateSpreadsheet_(props);
    var folder = getOrCreateFolder_(props);
    var leadsSheet = ensureSheet_(spreadsheet, 'Leads', LEAD_HEADERS);
    var runsSheet = ensureSheet_(spreadsheet, 'Runs', RUN_HEADERS);
    var prior = findRun_(runsSheet, interactionId);
    if (prior && prior.status === 'complete') {
      return json_({
        ok: true,
        duplicate: true,
        interactionId: interactionId,
        reportUrl: prior.reportUrl
      });
    }

    var parsed = parseLeadTable_(markdown);
    if (!parsed.foundTable) {
      return json_({ ok: false, error: 'No leads table found. Expected a markdown table whose header contains Company and Pain Point.' });
    }

    var leads = parsed.leads.slice(0, config.maxLeads);
    var reportUrl = archiveReport_(folder, interactionId, markdown);
    var home = config.homeBase ? geocodeAddress_(config.homeBase) : null;
    var summary = {
      ok: true,
      interactionId: interactionId,
      truncated: parsed.leads.length > leads.length,
      leadsReceived: parsed.leads.length,
      leadsWritten: 0,
      reportUrl: reportUrl,
      briefings: [],
      drafts: [],
      tasks: [],
      calendarUpdates: [],
      errors: []
    };

    var started = Date.now();
    for (var i = 0; i < leads.length; i += 1) {
      if (Date.now() - started > 4.5 * 60 * 1000) {
        summary.truncated = true;
        summary.errors.push({ company: '', notes: ['Stopped before the Apps Script time limit. Retry this delivery.'] });
        break;
      }
      var lead = leads[i];
      var outcome = processLead_(lead, {
        body: body,
        config: config,
        folder: folder,
        sheet: leadsSheet,
        home: home,
        markdown: markdown,
        reportUrl: reportUrl
      });
      summary.leadsWritten += 1;
      if (outcome.briefingUrl) summary.briefings.push(outcome.briefingUrl);
      if (outcome.draftId) summary.drafts.push(outcome.draftId);
      if (outcome.taskId) summary.tasks.push(outcome.taskId);
      if (outcome.calendarEventIds.length) {
        summary.calendarUpdates = summary.calendarUpdates.concat(outcome.calendarEventIds);
      }
      if (outcome.notes.length) {
        summary.errors.push({ company: lead.company, notes: outcome.notes });
      }
    }

    recordRun_(runsSheet, interactionId, summary.leadsWritten, reportUrl, 'complete');
    return json_(summary);
  } finally {
    lock.releaseLock();
  }
}

function processLead_(lead, ctx) {
  var notes = [];
  var blank = {
    briefingUrl: '',
    draftId: '',
    taskId: '',
    taskDue: '',
    calendarEventIds: [],
    latitude: '',
    longitude: '',
    distanceKm: '',
    notes: notes
  };

  if (leadExists_(ctx.sheet, ctx.body.interactionId, lead)) {
    notes.push('Row already existed; side effects were not repeated.');
    blank.notes = notes;
    return blank;
  }

  var geo = lead.hqAddress ? geocodeAddress_(lead.hqAddress) : { error: '' };
  if (geo.error) notes.push('Geocode: ' + geo.error);
  if (geo.lat != null) {
    blank.latitude = geo.lat;
    blank.longitude = geo.lng;
    if (ctx.home && ctx.home.lat != null) {
      blank.distanceKm = haversineKm_(ctx.home, geo);
    } else if (ctx.config.homeBase) {
      notes.push('Home base could not be geocoded, so proximity was skipped.');
    }
  }

  var highValue = isHighValue_(lead);
  if (highValue) {
    try {
      var briefing = createBriefingDoc_(ctx.folder, lead, ctx.markdown, geo, blank.distanceKm, ctx.body.interactionId);
      blank.briefingUrl = briefing.url;
      var calendar = attachBriefingToCalendar_(lead, briefing.url, ctx.config.lookaheadDays);
      blank.calendarEventIds = calendar.updated;
      if (calendar.note) notes.push(calendar.note);
    } catch (err) {
      notes.push('Briefing: ' + (err.message || err));
    }

    try {
      var task = createFollowUpTask_(lead, blank.briefingUrl, ctx.config.tasklist);
      blank.taskId = task.id;
      blank.taskDue = task.due;
    } catch (err) {
      notes.push('Task: ' + (err.message || err));
    }
    notes.push('Draft is not created until the pitch is approved. Nothing is sent from this desk.');
  } else {
    notes.push('Negative match. No briefing, task, or draft was created.');
  }

  appendLead_(ctx.sheet, lead, ctx.body, blank);
  return blank;
}

function readBody_(e) {
  var raw = e && e.postData && e.postData.contents;
  if (!raw) throw new Error('Missing JSON body.');
  try {
    return JSON.parse(raw);
  } catch (err) {
    throw new Error('Body is not valid JSON.');
  }
}

function providedSecret_(e) {
  var headers = (e && e.headers) || {};
  return String(headers['X-Webhook-Secret'] || headers['x-webhook-secret'] || '').trim();
}

function secretsMatch_(provided, expected) {
  if (!provided || !expected || provided.length !== expected.length) return false;
  var mismatch = 0;
  for (var i = 0; i < expected.length; i += 1) {
    mismatch |= provided.charCodeAt(i) ^ expected.charCodeAt(i);
  }
  return mismatch === 0;
}

function getConfig_(payload) {
  var props = PropertiesService.getScriptProperties();
  var seller = payload.seller || {};
  return {
    secret: String(props.getProperty('WEBHOOK_SECRET') || '').trim(),
    sellerName: seller.name || props.getProperty('SELLER_NAME') || '',
    sellerOffering: seller.offering || props.getProperty('SELLER_OFFERING') || '',
    homeBase: props.getProperty('HOME_BASE_ADDRESS') || '',
    lookaheadDays: clamp_(Number(props.getProperty('CALENDAR_LOOKAHEAD_DAYS') || 30), 1, 90),
    maxLeads: clamp_(Number(props.getProperty('MAX_LEADS') || 8), 1, HARD_LEAD_CAP),
    tasklist: props.getProperty('TASKLIST_ID') || '@default'
  };
}

function json_(value) {
  return ContentService.createTextOutput(JSON.stringify(value)).setMimeType(ContentService.MimeType.JSON);
}

function clamp_(value, min, max) {
  if (!isFinite(value)) return min;
  return Math.min(max, Math.max(min, Math.round(value)));
}

function getOrCreateSpreadsheet_(props) {
  var existing = props.getProperty('SPREADSHEET_ID');
  if (existing) return SpreadsheetApp.openById(existing);
  var created = SpreadsheetApp.create('Lead Generation');
  props.setProperty('SPREADSHEET_ID', created.getId());
  return created;
}

function getOrCreateFolder_(props) {
  var existing = props.getProperty('BRIEFINGS_FOLDER_ID');
  if (existing) return DriveApp.getFolderById(existing);
  var created = DriveApp.createFolder('Leads Briefings');
  props.setProperty('BRIEFINGS_FOLDER_ID', created.getId());
  return created;
}

var COLUMN_MAPS_ = {};

function ensureSheet_(spreadsheet, name, headers) {
  var sheet = spreadsheet.getSheetByName(name) || spreadsheet.insertSheet(name);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
  }
  rememberColumns_(sheet, headers);
  return sheet;
}

function rememberColumns_(sheet, headers) {
  var map = {};
  var width = sheet.getLastColumn();
  var current = width ? sheet.getRange(1, 1, 1, width).getValues()[0] : [];
  for (var i = 0; i < current.length; i += 1) {
    var name = String(current[i] || '').trim();
    if (name && map[name] == null) map[name] = i + 1;
  }
  var missing = headers.filter(function (header) { return map[header] == null; });
  if (missing.length) {
    var start = Math.max(width, 0) + 1;
    sheet.getRange(1, start, 1, missing.length).setValues([missing]).setFontWeight('bold');
    missing.forEach(function (header, index) { map[header] = start + index; });
  }
  COLUMN_MAPS_[String(sheet.getSheetId())] = map;
  return map;
}

function column_(sheet, header) {
  var map = COLUMN_MAPS_[String(sheet.getSheetId())] || {};
  return map[header] || 0;
}

function readCell_(row, sheet, header) {
  var index = column_(sheet, header) - 1;
  if (index < 0 || index >= row.length) return '';
  return row[index];
}

function sheetKey_(value) {
  var text = String(value == null ? '' : value).trim();
  if (text.charAt(0) === "'") text = text.slice(1).trim();
  return text.toLowerCase();
}

function writeMappedRow_(sheet, rowNumber, fields) {
  var max = 1;
  Object.keys(fields).forEach(function (header) {
    var column = column_(sheet, header);
    if (column > max) max = column;
  });
  var width = Math.max(sheet.getLastColumn(), max);
  var existing = rowNumber <= sheet.getLastRow() && width
    ? sheet.getRange(rowNumber, 1, 1, width).getValues()[0]
    : [];
  var row = [];
  for (var i = 0; i < width; i += 1) row.push(existing[i] == null ? '' : existing[i]);
  Object.keys(fields).forEach(function (header) {
    var column = column_(sheet, header);
    if (column) row[column - 1] = fields[header];
  });
  sheet.getRange(rowNumber, 1, 1, row.length).setValues([row]);
}

function findRun_(sheet, interactionId) {
  var values = sheet.getDataRange().getValues();
  var wanted = sheetKey_(interactionId);
  for (var i = 1; i < values.length; i += 1) {
    if (sheetKey_(readCell_(values[i], sheet, 'Interaction ID')) === wanted) {
      return {
        row: i + 1,
        reportUrl: readCell_(values[i], sheet, 'Report URL'),
        status: readCell_(values[i], sheet, 'Status')
      };
    }
  }
  return null;
}

function recordRun_(sheet, interactionId, leadCount, reportUrl, status) {
  var prior = findRun_(sheet, interactionId);
  var fields = {
    'Interaction ID': interactionId,
    'Processed At': new Date().toISOString(),
    'Lead Count': leadCount,
    'Report URL': reportUrl,
    'Status': status
  };
  writeMappedRow_(sheet, prior ? prior.row : sheet.getLastRow() + 1, fields);
}

function leadExists_(sheet, interactionId, lead) {
  var values = sheet.getDataRange().getValues();
  var wantedId = sheetKey_(interactionId);
  var wantedCompany = sheetKey_(lead.company);
  var wantedDomain = sheetKey_(lead.domain);
  for (var i = 1; i < values.length; i += 1) {
    var sameRun = sheetKey_(readCell_(values[i], sheet, 'Interaction ID')) === wantedId;
    var sameCompany = sheetKey_(readCell_(values[i], sheet, 'Company')) === wantedCompany;
    var sameDomain = wantedDomain && sheetKey_(readCell_(values[i], sheet, 'Domain')) === wantedDomain;
    if (sameRun && (sameCompany || sameDomain)) return true;
  }
  return false;
}

function appendLead_(sheet, lead, body, outcome) {
  writeMappedRow_(sheet, sheet.getLastRow() + 1, {
    'Generated At': body.generatedAt || new Date().toISOString(),
    'Interaction ID': body.interactionId,
    'Company': asCell_(lead.company),
    'Domain': asCell_(lead.domain),
    'HQ Address': asCell_(lead.hqAddress),
    'Latitude': outcome.latitude,
    'Longitude': outcome.longitude,
    'Distance Km': outcome.distanceKm,
    'Contact Name': asCell_(lead.contactName),
    'Contact Title': asCell_(lead.title),
    'Contact Email': asCell_(lead.email),
    'Pain Point': asCell_(lead.painPoint),
    'Hook': asCell_(lead.hook),
    'Priority': asCell_(lead.priority),
    'Source': asCell_(lead.source),
    'Briefing URL': outcome.briefingUrl,
    'Draft ID': outcome.draftId,
    'Calendar Event IDs': outcome.calendarEventIds.join(', '),
    'Task ID': outcome.taskId,
    'Task Due': outcome.taskDue,
    'Notes': asCell_(outcome.notes.join(' | '))
  });
}

function asCell_(value) {
  var text = value == null ? '' : String(value);
  return /^[=+\-@]/.test(text) ? "'" + text : text;
}

function parseLeadTable_(markdown) {
  var lines = String(markdown || '').split(/\r?\n/);
  var headerIndex = -1;
  for (var i = 0; i < lines.length; i += 1) {
    var candidate = lines[i].toLowerCase();
    if (candidate.indexOf('|') !== -1 && candidate.indexOf('company') !== -1 && candidate.indexOf('pain') !== -1) {
      headerIndex = i;
      break;
    }
  }
  if (headerIndex === -1) return { foundTable: false, leads: [] };

  var headers = splitRow_(lines[headerIndex]).map(normalizeHeader_);
  var leads = [];
  for (var row = headerIndex + 1; row < lines.length; row += 1) {
    var line = lines[row].trim();
    if (!line || line.charAt(0) !== '|') break;
    var cells = splitRow_(line);
    if (cells.every(function (cell) { return /^:?-{3,}:?$/.test(cell); })) continue;
    var raw = {};
    headers.forEach(function (header, index) {
      if (header) raw[header] = cells[index] || '';
    });
    var lead = canonicalLead_(raw);
    if (lead) leads.push(lead);
  }
  return { foundTable: true, leads: leads };
}

function splitRow_(line) {
  var trimmed = String(line || '').trim().replace(/^\|/, '').replace(/\|$/, '');
  var cells = [];
  var current = '';
  for (var i = 0; i < trimmed.length; i += 1) {
    var char = trimmed.charAt(i);
    if (char === '|' && trimmed.charAt(i - 1) !== '\\') {
      cells.push(current.replace(/\\\|/g, '|').trim());
      current = '';
    } else {
      current += char;
    }
  }
  cells.push(current.replace(/\\\|/g, '|').trim());
  return cells;
}

function normalizeHeader_(cell) {
  var cleaned = String(cell || '').replace(/\*\*/g, '').replace(/`/g, '').trim().toLowerCase();
  if (cleaned.indexOf('email') !== -1) return 'email';
  if (cleaned.indexOf('pain') !== -1) return 'painPoint';
  if (cleaned.indexOf('hook') !== -1) return 'hook';
  if (cleaned.indexOf('prior') !== -1) return 'priority';
  if (cleaned.indexOf('source') !== -1 || cleaned.indexOf('url') !== -1) return 'source';
  if (cleaned.indexOf('domain') !== -1 || cleaned.indexOf('website') !== -1) return 'domain';
  if (cleaned.indexOf('address') !== -1 || cleaned.indexOf('headquarter') !== -1 || cleaned === 'hq') return 'hqAddress';
  if (cleaned.indexOf('title') !== -1 || cleaned.indexOf('role') !== -1) return 'title';
  if (cleaned.indexOf('contact') !== -1) return 'contactName';
  if (cleaned.indexOf('company') !== -1 || cleaned.indexOf('account') !== -1) return 'company';
  return '';
}

function canonicalLead_(raw) {
  var company = cleanField_(raw.company);
  if (!company || /^(unknown|n\/a|none|-)$/i.test(company)) return null;
  var domain = normalizeDomain_(raw.domain);
  return {
    company: company,
    domain: domain === 'unknown' ? '' : domain,
    hqAddress: unknownToBlank_(raw.hqAddress),
    contactName: unknownToBlank_(raw.contactName),
    title: unknownToBlank_(raw.title),
    email: extractEmail_(raw.email),
    painPoint: unknownToBlank_(raw.painPoint),
    hook: unknownToBlank_(raw.hook),
    priority: cleanField_(raw.priority) || 'unknown',
    source: cleanField_(raw.source)
  };
}

function cleanField_(value) {
  return String(value || '').replace(/\s+/g, ' ').trim();
}

function unknownToBlank_(value) {
  var cleaned = cleanField_(value);
  return /^(unknown|n\/a|none|-)$/i.test(cleaned) ? '' : cleaned;
}

function normalizeDomain_(value) {
  return cleanField_(value)
    .toLowerCase()
    .replace(/^https?:\/\//, '')
    .replace(/^www\./, '')
    .split('/')[0]
    .replace(/[^a-z0-9.-]/g, '');
}

function extractEmail_(value) {
  var match = String(value || '').match(/[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}/i);
  return match ? match[0] : '';
}

function isHighValue_(lead) {
  return String(lead.priority || '').toLowerCase().indexOf('high') === 0;
}

function extractBriefing_(markdown, company) {
  var lines = String(markdown || '').split(/\r?\n/);
  var start = -1;
  var heading = new RegExp('^#{1,3}\\s+Briefing:\\s+' + escapeRegExp_(company) + '\\s*$', 'i');
  for (var i = 0; i < lines.length; i += 1) {
    if (heading.test(lines[i].trim())) {
      start = i + 1;
      break;
    }
  }
  if (start === -1) return '';
  var section = [];
  for (var j = start; j < lines.length; j += 1) {
    if (/^#{1,3}\s+/.test(lines[j])) break;
    section.push(lines[j]);
  }
  return section.join('\n').trim();
}

function attachBriefingToCalendar_(lead, docUrl, lookaheadDays) {
  var calendar = CalendarApp.getDefaultCalendar();
  var cursor = new Date();
  var end = new Date(cursor.getTime() + lookaheadDays * 24 * 60 * 60 * 1000);
  var updated = [];
  var matched = 0;
  var failures = 0;
  var examined = 0;

  while (cursor < end && updated.length < 5 && examined < 500) {
    var chunkEnd = new Date(Math.min(end.getTime(), cursor.getTime() + 7 * 24 * 60 * 60 * 1000));
    var events = calendar.getEvents(cursor, chunkEnd);
    for (var i = 0; i < events.length && updated.length < 5 && examined < 500; i += 1) {
      examined += 1;
      var event = events[i];
      if (!eventMatches_(event, lead.domain, lead.company)) continue;
      matched += 1;
      try {
        var description = event.getDescription() || '';
        if (description.indexOf(docUrl) !== -1) continue;
        event.setDescription(description + '\n\nLead briefing (' + lead.company + '): ' + docUrl);
        updated.push(event.getId());
      } catch (err) {
        failures += 1;
      }
    }
    cursor = chunkEnd;
  }

  var note = '';
  if (failures) note = 'Could not edit ' + failures + ' matching calendar event(s). They may be owned by someone else.';
  if (examined === 500) note = (note ? note + ' ' : '') + 'Calendar scan stopped at 500 events.';
  if (!note && matched > updated.length) note = 'Matched ' + matched + ' events and attached the briefing to the first ' + updated.length + '.';
  return { updated: updated, note: note };
}

function eventMatches_(event, domain, company) {
  var haystack = ((event.getTitle() || '') + '\n' + (event.getDescription() || '')).toLowerCase();
  if (company && haystack.indexOf(company.toLowerCase()) !== -1) return true;
  if (!domain) return false;
  try {
    return (event.getGuestList() || []).some(function (guest) {
      return String(guest.getEmail() || '').toLowerCase().endsWith('@' + domain);
    });
  } catch (err) {
    return false;
  }
}

function createOutreachDraft_(lead, seller) {
  var needsRecipient = !looksLikeEmail_(lead.email);
  var recipient = needsRecipient ? '' : lead.email;
  var subject = buildSubject_(lead);
  if (needsRecipient) {
    try {
      recipient = Session.getActiveUser().getEmail() || '';
    } catch (err) {
      recipient = '';
    }
    subject = '[Needs recipient] ' + subject;
  }

  var existingId = findDraftBySubject_(subject);
  if (existingId) return { id: existingId, needsRecipient: needsRecipient };

  try {
    var draft = GmailApp.createDraft(recipient || '', subject, buildDraftBody_(lead, seller));
    return { id: draft.getId(), needsRecipient: needsRecipient };
  } catch (err) {
    Logger.log(err);
    return { id: '', needsRecipient: needsRecipient };
  }
}

function findDraftBySubject_(subject) {
  var start = 0;
  while (start < 500) {
    var threads = GmailApp.search('in:drafts newer_than:30d', start, 50);
    if (!threads.length) return '';
    for (var i = 0; i < threads.length; i += 1) {
      var messages = threads[i].getMessages();
      for (var m = 0; m < messages.length; m += 1) {
        if (messages[m].getSubject() === subject) return messages[m].getId();
      }
    }
    if (threads.length < 50) return '';
    start += 50;
  }
  return '';
}

function buildSubject_(lead) {
  var focus = lead.painPoint || lead.hook;
  var short = focus.length > 70 ? focus.slice(0, 67) + '...' : focus;
  return lead.company + ': ' + short;
}

function buildDraftBody_(lead, seller) {
  var greeting = lead.contactName ? 'Hi ' + lead.contactName.split(' ')[0] + ',' : 'Hi,';
  var paragraphs = [
    greeting,
    'I was reading about ' + lead.company + ' and this stood out: ' + lead.hook,
    seller.offering
      ? 'We work with teams on that kind of problem. ' + seller.offering
      : 'I can send a one-page brief if this is a current problem.',
    'If this is relevant I can send the short brief. If it is not, no need to reply.'
  ];
  if (seller.name) paragraphs.push(seller.name);
  paragraphs.push('Draft only. Review the sources before sending.');
  return paragraphs.join('\n\n');
}

function findOpenTask_(tasklist, title) {
  var token = '';
  var seen = 0;
  while (seen < 1000) {
    var options = { showCompleted: false, maxResults: 100 };
    if (token) options.pageToken = token;
    var page = Tasks.Tasks.list(tasklist, options);
    var items = (page && page.items) || [];
    for (var i = 0; i < items.length; i += 1) {
      seen += 1;
      if (items[i].title === title) return items[i];
    }
    token = page && page.nextPageToken;
    if (!token || !items.length) return null;
  }
  return null;
}

function approveLead_(body, config) {
  var company = cleanField_(body.company);
  var interactionId = String(body.interactionId || '').trim();
  if (!company || !interactionId) {
    return json_({ ok: false, sent: false, error: 'company and interactionId are required. Nothing was saved in Gmail.' });
  }
  var props = PropertiesService.getScriptProperties();
  var sheet = ensureSheet_(getOrCreateSpreadsheet_(props), 'Leads', LEAD_HEADERS);
  var row = findLeadRow_(sheet, interactionId, company);
  if (!row) {
    return json_({ ok: false, sent: false, error: 'That lead is not on the sheet yet. Nothing was saved in Gmail.' });
  }
  var values = sheet.getRange(row, 1, 1, Math.max(sheet.getLastColumn(), 1)).getValues()[0];
  var hook = cleanField_(readCell_(values, sheet, 'Hook'));
  var priority = cleanField_(readCell_(values, sheet, 'Priority'));
  if (!hook || priority.toLowerCase().indexOf('high') !== 0) {
    return json_({ ok: false, sent: false, error: 'A negative match does not create a draft. Nothing was sent.' });
  }
  var existingDraft = cleanField_(readCell_(values, sheet, 'Draft ID'));
  if (existingDraft) {
    return json_({ ok: true, sent: false, duplicate: true, draftId: existingDraft, taskCleared: false });
  }
  var lead = {
    company: company,
    email: cleanField_(readCell_(values, sheet, 'Contact Email')),
    contactName: cleanField_(readCell_(values, sheet, 'Contact Name')),
    painPoint: cleanField_(readCell_(values, sheet, 'Pain Point')),
    hook: hook
  };
  var draft = createOutreachDraft_(lead, { name: config.sellerName, offering: config.sellerOffering });
  var draftColumn = column_(sheet, 'Draft ID');
  if (draftColumn && draft.id) sheet.getRange(row, draftColumn).setValue(asCell_(draft.id));
  var taskId = cleanField_(readCell_(values, sheet, 'Task ID'));
  var taskCleared = false;
  if (taskId && typeof Tasks !== 'undefined' && Tasks.Tasks) {
    try {
      Tasks.Tasks.patch({ status: 'completed' }, config.tasklist, taskId);
      taskCleared = true;
    } catch (err) {
      Logger.log(err);
    }
  }
  return json_({
    ok: true,
    sent: false,
    draftId: draft.id,
    taskCleared: taskCleared,
    needsRecipient: Boolean(draft.needsRecipient)
  });
}

function geocodeAddress_(address) {
  var cleaned = cleanField_(address);
  if (!cleaned || /^(unknown|n\/a|none)$/i.test(cleaned)) return { error: '' };
  var cache = CacheService.getScriptCache();
  var key = 'geo:' + fingerprint_(cleaned);
  var hit = cache.get(key);
  if (hit) return JSON.parse(hit);

  var response = Maps.newGeocoder().geocode(cleaned);
  if (!response || response.status === 'OVER_QUERY_LIMIT') {
    return { error: 'Maps geocoder quota is exhausted for today.' };
  }
  if (response.status !== 'OK' || !response.results || !response.results.length) {
    return { error: 'No geocode match (' + (response && response.status ? response.status : 'empty') + ').' };
  }
  var location = response.results[0].geometry.location;
  var point = { lat: location.lat, lng: location.lng };
  cache.put(key, JSON.stringify(point), 21600);
  return point;
}

function haversineKm_(from, to) {
  var toRad = function (degrees) { return degrees * Math.PI / 180; };
  var earthKm = 6371;
  var dLat = toRad(to.lat - from.lat);
  var dLng = toRad(to.lng - from.lng);
  var lat1 = toRad(from.lat);
  var lat2 = toRad(to.lat);
  var h = Math.pow(Math.sin(dLat / 2), 2)
    + Math.cos(lat1) * Math.cos(lat2) * Math.pow(Math.sin(dLng / 2), 2);
  return Math.round(earthKm * 2 * Math.atan2(Math.sqrt(h), Math.sqrt(1 - h)) * 10) / 10;
}

function fingerprint_(value) {
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, value, Utilities.Charset.UTF_8);
  return bytes.map(function (byte) {
    var unsigned = (byte + 256) % 256;
    return ('0' + unsigned.toString(16)).slice(-2);
  }).join('').slice(0, 32);
}

      items.push({ id: 'drive:' + file.getId(), name: file.getName(), kind: 'Drive', url: file.getUrl() });
      count += 1;
    }
  });
  return json_({ ok: true, items: items });

function readWorkspaceItem_(id) {
  if (id.indexOf('gmail:') === 0) {
    var thread = GmailApp.getThreadById(id.slice(6));
    if (!thread) return '';
    return thread.getMessages().slice(0, 3).map(function (message) {
      return message.getFrom() + '\n' + message.getSubject() + '\n' + message.getPlainBody();
    }).join('\n\n');
  }
  if (id.indexOf('doc:') === 0) return DocumentApp.openById(id.slice(4)).getBody().getText();
  if (id.indexOf('sheet:') === 0) {
    var sheet = SpreadsheetApp.openById(id.slice(6)).getSheets()[0];
    return sheet.getDataRange().getValues().slice(0, 40).map(function (row) { return row.join('\t'); }).join('\n');
  }
  if (id.indexOf('slide:') === 0) return slideText_(id.slice(6));
  if (id.indexOf('drive:') === 0) {
    var file = DriveApp.getFileById(id.slice(6));
    var mime = file.getMimeType();
    if (mime === 'application/vnd.google-apps.document') return DocumentApp.openById(file.getId()).getBody().getText();
    if (mime === 'application/vnd.google-apps.spreadsheet') return readWorkspaceItem_('sheet:' + file.getId());
    if (mime === 'application/vnd.google-apps.presentation') return slideText_(file.getId());
    if (file.getSize() > 500000) return file.getName() + ' is on Drive at ' + file.getUrl();
    return file.getBlob().getDataAsString().slice(0, 12000);
  }
  return '';
}

function slideText_(id) {
  var presentation = SlidesApp.openById(id);
  return presentation.getSlides().slice(0, 15).map(function (slide) {
    return slide.getPageElements().map(function (element) {
      try {
        if (element.getPageElementType() === SlidesApp.PageElementType.SHAPE) return element.asShape().getText().asString();
      } catch (err) {
        return '';
      }
      return '';
    }).join(' ');
  }).join('\n');
}

  }

**Health score: 100 / 100.**

## 1. Information architecture and agent status

A batch is one research run. The desk never replaces the page with a spinner. The phase row stays visible: Planning, Searching, Drafting, Syncing to Sheets.

| Pain points | Vulnerability list | Sourced pain, or a red negative match |
| Priority | Financial result table | High creates briefing, calendar link, and task. Low is a sheet row only |
| Territory | Map table, not the hazard globe | City group and coordinates |

The report is a split inside the result card. The left side is the markdown. The right drawer holds a source-mix bar chart and citation links. Bars are counts for this batch. Citations are real URLs from the report, opened in the same tab as normal links.

Bulk synthesis stays in the columns. The report opens for one company at a time, so a finished batch does not dump every briefing onto the home screen.

## 3. Human review

The Gmail draft is a textarea on the desk. Approving it marks the lead Draft ready and shows the toast that the draft was saved. The desk does not send mail. If the row is a negative match, Approve explains that no pitch was created.

Calendar and Tasks are not a second calendar UI. The Workspace table and the amber reason under the draft use the same sentence that Apps Script writes into the event description and the task notes. For a High row that sentence includes the 48-hour due date. For a negative match it says no task was created.

Quick actions sit on the lead card:

| Action | Result |
| --- | --- |
| Open report | One-company drawer |
| Re-research company | Queues a new batch and keeps the old row until the new sync |
| Approve pitch and clear task | Saves the draft and moves the card to Action required until the person sends it |
| Archive lead | Removes it from the active columns and does not delete Drive files |

## 4. Accessibility, color, and speed

Text on white uses `#202124`. Secondary text uses `#3c4043`, which stays above the WCAG 2.2 AA 4.5:1 ratio at 13px. Links and the avatar use `#174ea6` on white. High-intent tags use `#0d6b32` on `#e6f4ea`. Negative matches use `#8a1c1c` on `#fdecea`. Amber reasons use `#7a4e00` on `#fff8e1`. The bright `#0dd573` from the HR stylesheet is not used for text or small icons, because it fails on white.

The search pill and the High tag are the loudest elements. Tiles, columns, and tables stay neutral. Focus is a 2px `#174ea6` ring, or white on the dark gate. The phase pulse is the only motion, and it is disabled for reduced motion.

Type is Montserrat on the gate, Outfit for the desk wordmark, and Inter everywhere else. Titles are 18px semibold. Phase labels are 13px bold. Body copy is 14–16px at 1.5 line height.

Hazard and Alpha Agents stay out of the navigation, the tile grid, and the pipeline.

def main() -> int:
    logging.basicConfig(
        level=os.environ.get("LOG_LEVEL", "INFO").upper(),
        format="%(asctime)s %(levelname)s %(name)s %(message)s",
    )
    log = logging.getLogger("fallback")
    if os.environ.get("LEADGEN_ROUTE") == "paid":
        log.error("Refusing to route again from inside the paid Deep Research process.")
        return 1
    try:
        settings = Settings.from_env(ROOT)
        scan = scan_repo(
            ROOT,
            {
                "DEMO_MODE": settings.demo_mode,
                "BILLING_ENABLED": settings.billing_enabled,
                "FORCE_FREE_TIER": settings.force_free,
                "VERIFY_BILLING": settings.verify_billing,
            },
        )
        scan.write(ROOT / "output" / "repo-scan.json")
        log.info(
            "Scan: paid_entrypoint=%s env_files=%s database_hits=%s key_name_present=%s",
            scan.paid_entrypoint,

            len(scan.database_hits),
            scan.key_name_present,
        )
        decision = decide(settings, scan)
        for reason in decision.reasons:
            log.info("Route %s: %s", decision.route, reason)
        if decision.route == "paid":
            return _run_paid()
        if decision.route == "demo":
            run_demo(settings)
            return 0
        run_free(settings)
        return 0
    except (RouteError, PipelineError) as exc:
        log.error("%s", exc)
        return 1


def _run_paid() -> int:

    if not node:

        return 1

    env["LEADGEN_ROUTE"] = "paid"
    completed = subprocess.run([node, "index.js"], cwd=ROOT, env=env, check=False)
    return int(completed.returncode or 0)

from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path


def load_dotenv(path: Path) -> None:
    if not path.is_file():
        return
    for raw in path.read_text(encoding="utf-8").splitlines():
        line = raw.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        if line.startswith("export "):
            line = line[len("export ") :]
        key, value = line.split("=", 1)
        key = key.strip()
        if not key or key in os.environ:
            continue
        os.environ[key] = value.strip().strip('"').strip("'")


def flag(name: str) -> bool:
    return os.environ.get(name, "").strip().lower() in {"1", "true", "yes"}


def clean_text(value: str, limit: int) -> str:
    return " ".join(value.replace("\r", " ").replace("\n", " ").split())[:limit]


def is_placeholder(value: str) -> bool:
    lowered = value.lower()
    return any(token in lowered for token in ("your-", "paste-", "changeme"))

"""Choose paid Deep Research, the free Flash loop, or the local demo cache.

Google's API does not return a billing-account boolean for an AI Studio key.
BILLING_ENABLED is the explicit opt-in for the paid agent. A short Flash call can
veto that opt-in when the key is rejected or the error says billing is required.
It cannot prove that Deep Research is prepaid.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass

from fallback.config import Settings, is_placeholder
from fallback.scan import RepoScan

log = logging.getLogger("fallback.decide")


class RouteError(RuntimeError):
    pass


@dataclass(frozen=True)
class Decision:
    route: str
    reasons: list[str]


def decide(settings: Settings, scan: RepoScan) -> Decision:
    reasons: list[str] = []
    if settings.demo_mode:
        reasons.append("DEMO_MODE=true. Loading the local JSON lead cache and skipping Gemini.")
        return Decision("demo", reasons)

    _require_key(settings)

    if settings.force_free or not scan.paid_entrypoint or not settings.billing_enabled:
        if settings.force_free:
            reasons.append("FORCE_FREE_TIER=true.")
        if not scan.paid_entrypoint:
            reasons.append("The Deep Research interactions.create path was not found in index.js.")
        if not settings.billing_enabled:
            reasons.append("BILLING_ENABLED is not true, so the paid agent is not selected.")
        return Decision("free", reasons)

    if settings.verify_billing:
        status = probe_key(settings)
        reasons.append(f"Key probe returned {status}.")
        if status == "billing_blocked":
            reasons.append("The API reported a billing block. Falling back to Flash.")
            return Decision("free", reasons)
        if status == "invalid_key":
            raise RouteError("GEMINI_API_KEY was rejected by the Gemini API.")
        if status == "error":
            reasons.append("The billing probe failed. Falling back to Flash instead of starting Deep Research.")
            return Decision("free", reasons)
        if status == "sdk_missing":
            reasons.append("Python google-genai is not installed. The paid path will use the Node SDK.")

    reasons.append("BILLING_ENABLED=true and index.js still contains the Deep Research path.")
    return Decision("paid", reasons)


def _require_key(settings: Settings) -> None:
    if settings.key_usable:
        return
    if not settings.api_key or is_placeholder(settings.api_key):
        raise RouteError("Set GEMINI_API_KEY in .env, or set DEMO_MODE=true to use the local lead cache.")
    raise RouteError("GEMINI_API_KEY is empty.")


def probe_key(settings: Settings) -> str:
    try:
        from google import genai
        from google.genai import types
    except ImportError:
        log.warning("google-genai is not installed; skipping the Python billing probe.")
        return "sdk_missing"
    try:
        client = genai.Client(api_key=settings.api_key)
        client.models.generate_content(
            model=settings.model,
            contents="Reply with the single word ok.",
            config=types.GenerateContentConfig(max_output_tokens=16, temperature=0),
        )
        return "ok"
    except Exception as exc:  # noqa: BLE001 - the SDK raises several error types
        text = str(exc).lower()
        log.warning("Key probe failed: %s", _redact(text, settings.api_key))
        if any(token in text for token in ("api key", "api_key", "invalid", "unauthorized", "401", "permission denied")):
            if "billing" not in text:
                return "invalid_key"
        if any(token in text for token in ("billing", "prepay", "payment required", "quota")):
            return "billing_blocked"
        return "error"


def _redact(text: str, secret: str) -> str:
    if secret:
        text = text.replace(secret, "[redacted]")
    return text[:300]

def run_demo(settings: Settings) -> dict:
    payload = json.loads((settings.root / "fallback" / "demo_leads.json").read_text(encoding="utf-8"))
    leads = payload.get("leads") or []
    if not leads:
        raise PipelineError("fallback/demo_leads.json has no leads.")
    markdown = render_markdown(leads)
    record = _record(settings, "demo-cache", f"demo-{_stamp()}", "demo cache", markdown)
    finish(settings, record)
    return record


def run_free(settings: Settings) -> dict:
    if not settings.offering or is_placeholder(settings.offering) or not settings.industry or is_placeholder(settings.industry):
        raise PipelineError("Set SELLER_OFFERING and TARGET_INDUSTRY before a free-tier research run.")
    quota = GroundingQuota(
        settings.root / "output" / "grounding-quota.json",
        settings.model,
        settings.grounding_calls_per_run,
    )
    log.info("%s", quota.limits.note)
    client = _client(settings)
    strategies = plan_searches(client, settings)
    findings = search_strategies(client, settings, quota, strategies)
    extra = review_findings(client, settings, findings)
    if extra and quota.allow():
        log.info("Reviewer requested %s validation search(es).", len(extra))
        findings.extend(search_strategies(client, settings, quota, extra))
    elif extra:
        log.warning("Skipping validation searches because the free grounding cap is reached.")
    if not any(item.get("text") for item in findings):
        raise PipelineError("Google Search grounding returned no usable text.")
    markdown = synthesize(client, settings, findings)
    if not _has_lead_table(markdown):
        log.warning("Synthesis missed the lead table. Asking Flash to repair the markdown.")
        markdown = repair_table(client, settings, markdown)
    record = _record(settings, settings.model, f"free-{_stamp()}", _query_summary(settings), markdown)
    finish(settings, record)
    return record


def _focus_line(settings: Settings) -> str:
    if not settings.research_company:
        return ""
    domain = f" ({settings.research_domain})" if settings.research_domain else ""
    return f"\nFocus this run on one company: {settings.research_company}{domain}. Use public sources about that company."


def plan_searches(client, settings: Settings) -> list[str]:
    prompt = f"""
Break this ideal-customer search into exactly 3 public web search queries.
Industry: {settings.industry}
Region: {settings.region}
Offering: {settings.offering}
Ideal customer: {settings.icp}{_focus_line(settings)}

Return only JSON of the form {{"strategies": ["query 1", "query 2", "query 3"]}}.
Each query looks for companies and a public operating problem. Do not search for personal email addresses, phone numbers, or private profiles.
""".strip()
    try:
        raw = _generate(client, settings, prompt, grounded=False, max_tokens=800, temperature=0.2)
        parsed = _parse_json(raw)
        queries = [str(item).strip() for item in parsed.get("strategies", []) if str(item).strip()]
    except (PipelineError, json.JSONDecodeError, TypeError, AttributeError) as exc:
        log.warning("Planner output was unusable (%s). Using three local queries.", exc)
        queries = []
    queries = [query for query in queries if not _looks_like_contact_harvest(query)][:3]
    if len(queries) < 3:
        queries = _local_queries(settings)
    log.info("Search strategies: %s", queries)
    return queries[:3]


def search_strategies(client, settings: Settings, quota: GroundingQuota, queries: list[str]) -> list[dict]:
    findings = []
    for query in queries:
        if not quota.allow():
            log.warning("Stopping search. Free grounding cap reached (%s this run).", quota.used_this_run)
            break
        prompt = (
            "Use Google Search to find public company information for this query. "
            "Summarize only companies a member of the public could identify. "
            "If a fact is not on the open web, say unknown. Do not invent email addresses. "
            f"Return at most {settings.max_leads} companies.\n\nQuery: {query}"
        )
        try:
            text, sources = _grounded_generate(client, settings, quota, prompt)
        except QuotaExceeded as exc:
            log.warning("%s", exc)
            break
        except PipelineError as exc:
            log.warning("Search failed for %r: %s", query, exc)
            continue
        findings.append({"query": query, "text": text, "sources": sources})
    return findings


def review_findings(client, settings: Settings, findings: list[dict]) -> list[str]:
    if not findings:
        return []
    prompt = f"""
You are reviewing public research notes for a lead list.
Decide whether the notes already contain enough distinct companies to stop, or whether one or two more public searches are required.
Maximum companies wanted: {settings.max_leads}

Notes:
{_notes(findings)}

Return only JSON: {{"enough": true, "validation_queries": []}}
Use at most 2 validation queries. Leave the array empty when enough is true.
Do not request searches for personal emails or private profiles.
""".strip()
    try:
        raw = _generate(client, settings, prompt, grounded=False, max_tokens=600, temperature=0.2)
        parsed = _parse_json(raw)
    except (PipelineError, json.JSONDecodeError, TypeError, AttributeError) as exc:
        log.warning("Reviewer output was unusable (%s). Stopping after the first searches.", exc)
        return []
    if parsed.get("enough") is True:
        log.info("Reviewer accepted the research notes.")
        return []
    queries = [str(item).strip() for item in parsed.get("validation_queries") or [] if str(item).strip()]
    return [query for query in queries if not _looks_like_contact_harvest(query)][:2]


def synthesize(client, settings: Settings, findings: list[dict]) -> str:
    prompt = f"""
Turn these public research notes into a lead list for this offering: {settings.offering}
Ideal customer: {settings.icp}{_focus_line(settings)}
Maximum rows: {settings.max_leads}

Notes:
{_notes(findings)}

Rules:
- Use only facts present in the notes. If a field is missing, write exactly "unknown".
- Never invent an email address, phone number, or quotation.
- Mark Priority as High only when both the pain point and the hook are tied to a public source in the notes. Otherwise use Medium or Low.

Start with this GitHub-flavored table and no prose before it:
{TABLE_HEADER}

After the table, for every High priority company, add:
## Briefing: {{Company}}
Three short paragraphs: what the company does, the sourced pain point, and the public hook.
""".strip()
    return _generate(client, settings, prompt, grounded=False, max_tokens=4096, temperature=0.2)


def repair_table(client, settings: Settings, markdown: str) -> str:
    prompt = (
        "Rewrite the following as the required lead table. Do not add companies or emails that are not already present. "
        "Missing fields stay unknown.\n\n"
        f"{TABLE_HEADER}\n\n{markdown[:12000]}"
    )
    return _generate(client, settings, prompt, grounded=False, max_tokens=4096, temperature=0)


def finish(settings: Settings, record: dict) -> str:
    output = settings.output_dir
    output.mkdir(parents=True, exist_ok=True)
    markdown_path = output / "last-report.md"
    meta_path = output / "last-report.json"
    markdown_path.write_text(record["markdown"], encoding="utf-8")
    meta_path.write_text(
        json.dumps(
            {
                "interactionId": record["interactionId"],
                "agent": record["agent"],
                "generatedAt": record["generatedAt"],
                "query": record["query"],
                "seller": {"name": settings.seller_name, "offering": settings.offering},
                "route": record["agent"],
            },
            indent=2,
        ),
        encoding="utf-8",
    )
    log.info("Saved report (%s chars) to %s", len(record["markdown"]), markdown_path)
    if not _has_lead_table(record["markdown"]):
        raise PipelineError("Report has no lead table. The webhook was not called.")
    if settings.webhook_url and not is_placeholder(settings.webhook_url):
        post_webhook(settings, record)
    else:
        reason = (
            "GOOGLE_WEBHOOK_URL is still a placeholder. Nothing was written to Sheets."
            if settings.webhook_url
            else "GOOGLE_WEBHOOK_URL is empty. Nothing was written to Sheets."
        )
        _write_webhook_result(settings, {"ok": False, "synced": False, "error": reason})
        log.info(reason)
    return str(markdown_path)


def _write_webhook_result(settings: Settings, payload: dict) -> None:
    settings.output_dir.mkdir(parents=True, exist_ok=True)
    safe = {key: value for key, value in payload.items() if key != "secret"}
    (settings.output_dir / "last-webhook.json").write_text(json.dumps(safe, indent=2), encoding="utf-8")


class _ManualRedirect(urllib.request.HTTPRedirectHandler):
    """Leave 3xx responses in place so the caller can POST again with the same body."""

    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None

    def http_error_302(self, req, fp, code, msg, headers):
        return fp

    http_error_301 = http_error_303 = http_error_307 = http_error_308 = http_error_302


def _post_preserving_redirect(url: str, body: bytes, headers: dict, timeout: int, hops: int = 5) -> tuple[int, str]:
    opener = urllib.request.build_opener(_ManualRedirect)
    current = url
    for _hop in range(hops):
        request = urllib.request.Request(current, data=body, headers=headers, method="POST")
        try:
            with opener.open(request, timeout=timeout) as response:
                status = getattr(response, "status", None) or response.getcode()
                raw = response.read().decode("utf-8", errors="replace")
                location = response.headers.get("Location")
        except urllib.error.HTTPError as exc:
            status = exc.code
            raw = exc.read().decode("utf-8", errors="replace")
            location = exc.headers.get("Location") if exc.headers else ""
        if status in (301, 302, 303, 307, 308):
            if not location:
                raise PipelineError("Apps Script redirected without a new address.")
            current = urllib.parse.urljoin(current, location)
            continue
        return status, raw
    raise PipelineError("Apps Script redirected too many times.")


def post_webhook(settings: Settings, record: dict) -> None:
    if not settings.webhook_url.startswith("https://") or is_placeholder(settings.webhook_url):
        raise PipelineError("GOOGLE_WEBHOOK_URL must be the https Apps Script /exec URL.")
    if len(settings.webhook_secret) < 16 or is_placeholder(settings.webhook_secret):
        raise PipelineError("Set WEBHOOK_SECRET to at least 16 characters before posting.")
    body = json.dumps(
        {
            "interactionId": record["interactionId"],
            "agent": record["agent"],
            "generatedAt": record["generatedAt"],
            "query": record["query"],
            "seller": {"name": settings.seller_name, "offering": settings.offering},
            "markdown": record["markdown"],
        }
    ).encode("utf-8")
    headers = {"Content-Type": "application/json", "X-Webhook-Secret": settings.webhook_secret}
    delay = 1.0
    last_error: Exception | None = None
    for attempt in range(1, 5):
        try:
            status, raw = _post_preserving_redirect(
                settings.webhook_url,
                body,
                headers,
                settings.request_timeout_seconds,
            )
            payload = json.loads(raw)
            if status >= 400 or payload.get("ok") is False:
                raise PipelineError(str(payload.get("error") or f"Webhook failed with HTTP {status}"))
            _write_webhook_result(
                settings,
                {"ok": True, "synced": True, "interactionId": record["interactionId"], "appsScript": payload},
            )
            log.info("Webhook accepted the report: %s", json.dumps({k: payload[k] for k in payload if k != "secret"}))
            return
        except (urllib.error.URLError, TimeoutError, json.JSONDecodeError, PipelineError) as exc:
            last_error = exc
            retryable = _retryable(exc)
            if not retryable or attempt == 4:
                break
            log.warning("Webhook attempt %s failed: %s", attempt, exc)
            time.sleep(delay)
            delay = min(16.0, delay * 2)
    _write_webhook_result(settings, {"ok": False, "synced": False, "error": f"Webhook failed: {last_error}"})
    raise PipelineError(f"Webhook failed: {last_error}")

def _generate(client, settings: Settings, prompt: str, *, grounded: bool, max_tokens: int, temperature: float) -> str:
    from google.genai import types

    config_kwargs = {"max_output_tokens": max_tokens, "temperature": temperature}
    if grounded:
        config_kwargs["tools"] = [types.Tool(google_search=types.GoogleSearch())]
        config_kwargs["temperature"] = 1.0
    delay = 2.0
    last_error: Exception | None = None
    for attempt in range(1, 4):
        try:
            response = client.models.generate_content(
                model=settings.model,
                contents=prompt,
                config=types.GenerateContentConfig(**config_kwargs),
            )
            text = (getattr(response, "text", None) or "").strip()
            if not text:
                raise PipelineError("The model returned an empty response.")
            return text
        except PipelineError:
            raise
        except Exception as exc:  # noqa: BLE001
            last_error = exc
            if not _retryable(exc) or attempt == 3:
                break
            log.warning("Model call failed (%s). Retrying in %.0fs.", exc, delay)
            time.sleep(delay)
            delay = min(20.0, delay * 2)

def _grounded_generate(client, settings: Settings, quota: GroundingQuota, prompt: str) -> tuple[str, list[str]]:
    from google.genai import types

    delay = 2.0
    last_error: Exception | None = None
    for attempt in range(1, 4):
        if not quota.allow():
            raise QuotaExceeded(quota.limits.note)
        try:
            response = client.models.generate_content(
                model=settings.model,
                contents=prompt,
                config=types.GenerateContentConfig(
                    tools=[types.Tool(google_search=types.GoogleSearch())],
                    temperature=1.0,
                    max_output_tokens=2048,
                ),
            )
            quota.commit()
            text = (getattr(response, "text", None) or "").strip()
            if not text:
                raise PipelineError("Grounded search returned an empty response.")
            return text, _sources(response)
        except QuotaExceeded:
            raise
        except Exception as exc:  # noqa: BLE001
            last_error = exc
            if _quota_exhausted(exc):
                quota.exhaust_today()
                raise QuotaExceeded(str(exc)) from exc
            if not _retryable(exc) or attempt == 3:
                break
            log.warning("Grounded search failed (%s). Retrying in %.0fs.", exc, delay)
            time.sleep(delay)
            delay = min(20.0, delay * 2)
    raise PipelineError(str(last_error or "grounded search failed"))


def _sources(response) -> list[str]:
    urls: list[str] = []
    candidates = getattr(response, "candidates", None) or []
    if not candidates:
        return urls
    metadata = getattr(candidates[0], "grounding_metadata", None)
    chunks = getattr(metadata, "grounding_chunks", None) or []
    for chunk in chunks:
        web = getattr(chunk, "web", None)
        uri = getattr(web, "uri", None)
        if uri:
            urls.append(str(uri))
    return urls[:8]


def _parse_json(text: str) -> dict:
    cleaned = text.strip()
    if cleaned.startswith("```"):
        cleaned = re.sub(r"^```(?:json)?\s*", "", cleaned)
        cleaned = re.sub(r"\s*```$", "", cleaned)
    try:
        value = json.loads(cleaned)
    except json.JSONDecodeError:
        match = re.search(r"\{.*\}", cleaned, re.DOTALL)
        if not match:
            raise
        value = json.loads(match.group(0))
    if not isinstance(value, dict):
        raise PipelineError("Model JSON was not an object.")
    return value

"""Local counter for Gemini Search grounding so one run cannot spend the free allotment.

Published Gemini Developer API limits (confirm on the pricing page before raising them):
- gemini-2.5-flash and Flash-Lite share 500 grounded requests per day on the free tier.
- Gemini 3 and newer models share 5,000 grounded requests per month, with a 1,500 request daily cap.
gemini-1.5-flash is treated with the stricter daily cap. This file cannot see usage from other apps.
"""

from __future__ import annotations

import json
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
from zoneinfo import ZoneInfo

QUOTA_TZ = ZoneInfo("America/Los_Angeles")


class QuotaExceeded(RuntimeError):
    pass


@dataclass
class ModelQuota:
    daily_cap: int
    monthly_cap: int | None
    note: str


def limits_for(model: str) -> ModelQuota:
    name = model.lower()
    if "gemini-3" in name:
        return ModelQuota(
            daily_cap=1500,
            monthly_cap=5000,
            note="Gemini 3 free grounding is 5,000 requests per month and 1,500 per day, shared across Gemini 3 models.",
        )
    return ModelQuota(
        daily_cap=500,
        monthly_cap=None,
        note="gemini-2.5-flash free grounding is 500 requests per day, shared with Flash-Lite. The 5,000/month figure is for Gemini 3 and newer, so it is not applied here.",
    )


class GroundingQuota:
    def __init__(self, path: Path, model: str, per_run_cap: int) -> None:
        self.path = path
        self.model = model
        self.limits = limits_for(model)
        self.per_run_cap = per_run_cap
        self.used_this_run = 0
        self._load()

    def allow(self) -> bool:
        under_month = self.limits.monthly_cap is None or self.month_count < self.limits.monthly_cap
        return self.used_this_run < self.per_run_cap and self.day_count < self.limits.daily_cap and under_month

    def commit(self) -> None:
        if not self.allow():
            raise QuotaExceeded(self.limits.note)
        self.used_this_run += 1
        self.day_count += 1
        self.month_count += 1
        self._save()

    def exhaust_today(self) -> None:
        self.day_count = self.limits.daily_cap
        self._save()

    def _load(self) -> None:
        now = datetime.now(QUOTA_TZ)
        self.day_key = now.strftime("%Y-%m-%d")
        self.month_key = now.strftime("%Y-%m")
        self.day_count = 0
        self.month_count = 0
        if not self.path.is_file():
            return
        try:
            saved = json.loads(self.path.read_text(encoding="utf-8"))
        except (OSError, json.JSONDecodeError):
            return
        if saved.get("day") == self.day_key:
            self.day_count = int(saved.get("day_count") or 0)
        if saved.get("month") == self.month_key:
            self.month_count = int(saved.get("month_count") or 0)

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(
            json.dumps(
                {
                    "model": self.model,
                    "day": self.day_key,
                    "day_count": self.day_count,
                    "month": self.month_key,
                    "month_count": self.month_count,
                    "daily_cap": self.limits.daily_cap,
                    "monthly_cap": self.limits.monthly_cap,
                },
                indent=2,
            ),
            encoding="utf-8",
        )

def scan_repo(root: Path, active_flags: dict[str, bool]) -> RepoScan:
    scan = RepoScan(root=str(root), active_flags=dict(active_flags))
    index_text = ""
    for path in _files(root):
        relative = path.relative_to(root).as_posix()
        if path.name in {".env", ".env.example"} or path.suffix == ".env":
            scan.env_files.append(relative)
            scan.env_key_names.extend(_env_key_names(path))
        if path.name in {"index.js", "Code.gs", "package.json", "appsscript.json"}:
            scan.logic_files.append(relative)
        if path.name == "index.js":
            index_text = path.read_text(encoding="utf-8", errors="ignore")
        if path.name != "scan.py" and path.suffix.lower() in TEXT_SUFFIXES and path.name != ".env":
            scan.database_hits.extend(_database_hits(path, relative))
    scan.env_key_names = sorted(set(scan.env_key_names))
    scan.key_name_present = "GEMINI_API_KEY" in scan.env_key_names
    scan.paid_entrypoint = "interactions.create" in index_text and "deep-research" in index_text
    return scan


def _files(root: Path):
    for path in root.rglob("*"):
        if not path.is_file():
            continue
        if any(part in SKIP_DIRS for part in path.parts):
            continue
        if path.stat().st_size > 1_000_000:
            continue
        yield path


def _env_key_names(path: Path) -> list[str]:
    names = []
    for raw in path.read_text(encoding="utf-8", errors="ignore").splitlines():
        match = ENV_KEY.match(raw)
        if match:
            names.append(match.group(1))
    return names


def _database_hits(path: Path, relative: str) -> list[dict[str, str]]:
    try:
        text = path.read_text(encoding="utf-8", errors="ignore")
    except OSError:
        return []
    hits = []
    for name, pattern in DB_PATTERNS:
        if pattern.search(text):
            hits.append({"file": relative, "kind": name})
    return hits

function assertConfig(config) {
  const needsGemini = !config.dryRun && !config.reportFile;
  if (needsGemini && (!config.apiKey || isPlaceholder(config.apiKey))) {

  if (needsGemini && !config.resumeId && (!config.offering || isPlaceholder(config.offering))) {
    throw new Error('Set SELLER_OFFERING in .env before starting research.');

  if (needsGemini && !config.resumeId && (!config.industry || isPlaceholder(config.industry))) {
    throw new Error('Set TARGET_INDUSTRY in .env before starting research.');
  }
  if (config.pollMaxDelayMs < config.pollInitialDelayMs) {
    throw new Error('POLL_MAX_DELAY_MS must be greater than or equal to POLL_INITIAL_DELAY_MS.');
  }
  if (!config.webhookUrl) {
    if (!config.dryRun) {
      throw new Error('Set GOOGLE_WEBHOOK_URL to the Apps Script deployment URL ending in /exec.');
    }
    return;
  }
  let parsed;
  try {
    parsed = new URL(config.webhookUrl);
  } catch {
    throw new Error('GOOGLE_WEBHOOK_URL is not a valid URL.');
  }
  if (parsed.protocol !== 'https:' || isPlaceholder(config.webhookUrl)) {
    throw new Error('GOOGLE_WEBHOOK_URL must be the https Apps Script /exec URL from your deployment.');
  }
  if (!config.webhookUrl.endsWith('/exec')) {
    console.warn('GOOGLE_WEBHOOK_URL does not end in /exec. Apps Script accepts POSTs on the deployment URL, not /dev.');
  }
  if (!config.webhookSecret || isPlaceholder(config.webhookSecret) || config.webhookSecret.length < 16) {
    throw new Error('Set WEBHOOK_SECRET to a random string of at least 16 characters, matching the Apps Script property.');
  }
}

function extractReport(interaction) {
  const chunks = [];
  const steps = Array.isArray(interaction.steps) ? interaction.steps : [];
  const modelSteps = steps.filter((step) => step && (step.type === 'model_output' || step.type === 'output'));
  const chosen = modelSteps.at(-1) || steps.at(-1);
  if (chosen) chunks.push(...textsFromContent(chosen.content));

  if (!chunks.length && Array.isArray(interaction.outputs)) {
    for (const output of interaction.outputs) {
      if (typeof output === 'string') chunks.push(output);
      else if (typeof output?.text === 'string') chunks.push(output.text);
      else chunks.push(...textsFromContent(output?.content));
    }
  }

  const markdown = chunks.map((chunk) => chunk.trim()).filter(Boolean).join('\n\n').trim();
  if (!markdown) {
    throw new Error(`Interaction ${interaction.id || 'unknown'} completed with no text output.`);
  }
  return markdown;
}

function assertLeadTable(markdown) {
  const found = markdown.split(/\r?\n/).some((line) => {
    const normalized = line.toLowerCase();
    return normalized.includes('|') && normalized.includes('company') && normalized.includes('pain');
  });
  if (!found) {
    throw new Error('Report has no lead table. The markdown is in output/last-report.md. Fix or replay it with REPORT_FILE. The webhook was not called.');
  }
}

async function pollInteraction(client, interactionId, config) {
  const started = Date.now();
  let delay = config.pollInitialDelayMs;
  let transientFailures = 0;
  let continuations = 0;
  let currentId = interactionId;

  while (true) {
    if (Date.now() - started > config.pollMaxWaitMs) {
      throw new Error(`Polling timed out after ${config.pollMaxWaitMs}ms. Resume with RESUME_INTERACTION_ID=${currentId}`);
    }

    let interaction;
    try {
      interaction = await client.interactions.get(currentId);
      transientFailures = 0;
    } catch (error) {
      transientFailures += 1;
      if (!isRetryable(error) || transientFailures > config.pollMaxTransientErrors) throw error;
      console.warn(`Poll error ${transientFailures}/${config.pollMaxTransientErrors}: ${formatError(error)}`);
      await sleep(delay);
      delay = nextDelay(delay, config);
      continue;
    }

    const status = String(interaction.status || '').toLowerCase();
    const elapsed = Math.round((Date.now() - started) / 1000);
    console.log(`status=${status} elapsed=${elapsed}s nextDelayMs=${delay} id=${currentId}`);

    if (status === 'completed') return interaction;
    if (status === 'failed' || status === 'cancelled') {
      throw new Error(`Interaction ${currentId} ${status}: ${formatError(interaction.error)}`);
    }

    if (status === 'requires_continuation') {
      if (!interaction.continuation_token || continuations >= 3) {
        throw new Error(`Interaction ${currentId} requires continuation and cannot be resumed automatically.`);
      }
      continuations += 1;
      const continued = await client.interactions.create({
        agent: config.agent,
        background: true,
        store: true,
        continuation_token: interaction.continuation_token,
        agent_config: AGENT_CONFIG,
      });
      if (!continued?.id) {
        throw new Error(`Continuation ${continuations} did not return an interaction id.`);
      }
      currentId = continued.id;
      activeInteractionId = currentId;
      console.log(`Continued research as ${currentId}`);
      continue;
    }

    await sleep(delay);
    delay = nextDelay(delay, config);
  }
}

async function runResearch(config) {
  const client = await createClient(config.apiKey);
  let interactionId = config.resumeId;
  const prompt = buildLeadPrompt(config);

  if (!interactionId) {
    const created = await client.interactions.create({
      input: prompt,
      agent: config.agent,
      background: true,
      store: true,
      agent_config: AGENT_CONFIG,
    });
    if (!created?.id) throw new Error('interactions.create did not return an interaction id.');
    interactionId = created.id;
    console.log(`Research started. Interaction ID: ${interactionId}`);
  } else {
    console.log(`Resuming interaction ${interactionId}`);
  }

  activeInteractionId = interactionId;
  const completed = await pollInteraction(client, interactionId, config);
  return {
    interactionId,
    agent: config.agent,
    generatedAt: new Date().toISOString(),
    query: config.resumeId ? `resume:${config.resumeId}` : prompt,
    markdown: extractReport(completed),
  };
}

async function postWebhook(config, payload) {
  let delay = 1000;
  let lastError;

  for (let attempt = 1; attempt <= 4; attempt += 1) {
    try {
      const outcome = await postAppsScript(config.webhookUrl, config.webhookSecret, payload, config.requestTimeoutMs);
      const body = outcome.body || {};
      if (outcome.status >= 500 || body.ok === false) {
        const error = new Error(body.error || `Webhook failed with HTTP ${outcome.status}`);
        error.retryable = Boolean(body.retryable) || outcome.status === 429 || outcome.status >= 500;
        throw error;
      }
      return body;
    } catch (error) {
      lastError = error;
      const retryable = Boolean(error.retryable) || isRetryable(error);
      if (!retryable || attempt === 4) break;
      console.warn(`Webhook attempt ${attempt} failed: ${error.message}. Retrying in ${delay}ms.`);
      await sleep(delay);
      delay = Math.min(16_000, delay * 2);
    }
  }

  throw lastError;
}

function writeWebhookResult(config, payload) {

  delete safe.secret;

}

function delegateToFallback() {
  const candidates = process.platform === 'win32' ? ['py', 'python', 'python3'] : ['python3', 'python'];
  for (const command of candidates) {
    const result = spawnSync(command, ['-m', 'fallback'], {
      cwd: process.cwd(),
      stdio: 'inherit',
      env: process.env,
    });
    if (result.error && result.error.code === 'ENOENT') continue;
    if (result.error) throw result.error;
    if (result.status !== 0) {
      throw new Error(`Free-tier fallback exited with status ${result.status ?? 'unknown'}.`);
    }
    return;
  }
  throw new Error('Python was not found on PATH. Install Python 3.11 or newer to use the free-tier fallback.');
}

const HEADER = '| Company | Domain | HQ Address | Contact Name | Contact Title | Contact Email | Pain Point | Hook | Priority | Source |';

export function parseLeadAnswer(text) {
  const raw = String(text || '');
  const explanationMatch = raw.match(/EXPLANATION:\s*([\s\S]*?)(?:\n\s*LEADS:|$)/i);
  const explanation = (explanationMatch ? explanationMatch[1] : raw).trim().slice(0, 2000);
  const lines = raw.split(/\r?\n/).map((line) => line.trim()).filter((line) => line.startsWith('|'));
  const leads = [];
  if (lines.length >= 2) {
    const headers = splitRow(lines[0]);
    const column = (label) => headers.findIndex((cell) => cell.toLowerCase().includes(label));
    const cols = {
      company: column('company'),
      domain: column('domain'),
      hq: column('hq'),
      contact: column('contact name'),
      title: column('contact title'),
      email: column('email'),
      pain: column('pain'),
      hook: column('hook'),
      priority: column('priority'),
      source: column('source'),
    };
    for (const line of lines.slice(1)) {
      const cells = splitRow(line);
      if (cells.every((cell) => /^:?-+:?$/.test(cell))) continue;
      const company = cells[cols.company] || '';
      if (!company) continue;
      let email = cells[cols.email] || '';
      if (!email.includes('@') || /unknown|n\/a|none|not on file|no public/i.test(email)) email = '';
      const priority = /high/i.test(cells[cols.priority] || '') && (cells[cols.hook] || '').trim() ? 'High' : 'Low';
      leads.push({
        company,
        domain: cells[cols.domain] || '',
        hq: cells[cols.hq] || '',
        contact: cells[cols.contact] || '',
        title: cells[cols.title] || '',
        email,
        pain: cells[cols.pain] || '',
        hook: priority === 'High' ? (cells[cols.hook] || '') : '',
        priority,
        source: cells[cols.source] || '',
      });
    }
  }
  return { explanation, leads: leads.slice(0, 15) };
}

    profile.TARGET_REGION ? `Region: ${profile.TARGET_REGION}` : '',

export async function askSources({ apiKey, model, question, profile, parts, urls, tabular }) {
  const { GoogleGenAI } = await import('@google/genai');
  const ai = new GoogleGenAI({ apiKey });
  const content = [{ text: askPrompt(question, profile) }, ...parts];
  const toolSets = [];
  const rich = [{ googleSearch: {} }];
  if (urls) rich.push({ urlContext: {} });
  if (tabular) rich.push({ codeExecution: {} });
  toolSets.push(rich);
  if (rich.length > 1) toolSets.push([{ googleSearch: {} }]);
  toolSets.push([]);
  let lastError = '';
  for (const tools of toolSets) {
    try {
      const response = await ai.models.generateContent({
        model,
        contents: [{ role: 'user', parts: content }],
        config: tools.length ? { tools } : undefined,
      });
      const text = responseText(response);
      if (text.trim()) return parseLeadAnswer(text);
    } catch (error) {
      lastError = error instanceof Error ? error.message : 'Gemini refused the request.';
    }
  }
  throw new Error(lastError || 'Gemini returned no text. Nothing was written to Sheets.');
}

function responseText(response) {
  try {
    return String(response?.text || '');
  } catch {
    return '';
  }
}

const memory = new Map();

function storePath(root) {
  return path.join(root, 'output', 'discovered-leads.json');
}

function load(root) {
  const key = path.resolve(root);
  if (loaded.has(key)) return;
  loaded.add(key);
  try {
    const parsed = JSON.parse(readFileSync(storePath(key), 'utf8'));
    for (const lead of parsed.leads || []) {
      const normalized = normalizeLead(lead);
      memory.set(`${key}:${normalized.id}`, normalized);
    }
  } catch {
    // A missing store means only the demo rows are available.
  }
}

function persist(root) {
  const key = path.resolve(root);
  const leads = [];
  for (const [id, lead] of memory) {
    if (id.startsWith(`${key}:`)) leads.push(lead);
  }
  mkdirSync(path.join(key, 'output'), { recursive: true });
  writeFileSync(storePath(key), JSON.stringify({ leads }, null, 2));
}

export function rememberLeads(root, incoming) {
  load(root);
  const key = path.resolve(root);
  const stored = [];
  for (const raw of incoming || []) {
    const lead = normalizeLead(raw);
    memory.set(`${key}:${lead.id}`, lead);
    stored.push(lead);
  }
  if (stored.length) persist(root);
  return stored;
}

  load(root);

  return [...merged.values()];

import { inflateRawSync } from 'node:zlib';

const CAP = 80000;

export function officeText(buffer) {
  const entries = zipEntries(buffer);
  if (entries.length) {
    const text = officeFromZip(entries);
    if (text) return text.slice(0, CAP);
  }
  return binaryStrings(buffer).slice(0, CAP);
}

function officeFromZip(entries) {
  const files = new Map(entries.map((entry) => [entry.name.replace(/\\/g, '/').toLowerCase(), entry.text]));
  const sheets = [...files.keys()].filter((name) => /^xl\/worksheets\/sheet\d+\.xml$/.test(name)).sort();
  if (sheets.length) {
    const shared = sharedStrings(files.get('xl/sharedstrings.xml') || '');
    return sheets.map((name) => sheetTable(files.get(name), shared)).filter(Boolean).join('\n\n');
  }
  const parts = [...files.entries()]
    .filter(([name]) => name.endsWith('document.xml') || name === 'content.xml' || /ppt\/slides\/slide\d+\.xml$/.test(name) || /word\/header\d+\.xml$/.test(name) || /word\/footer\d+\.xml$/.test(name))
    .map(([, text]) => xmlText(text));
  return parts.join('\n').replace(/[ \t]+\n/g, '\n').replace(/\n{3,}/g, '\n\n').trim();
}

function sharedStrings(xml) {
  return [...String(xml).matchAll(/<si\b[^>]*>([\s\S]*?)<\/si>/g)].map((match) =>
    [...match[1].matchAll(/<t\b[^>]*>([\s\S]*?)<\/t>/g)].map((part) => decodeXml(part[1])).join('')
  );
}

function sheetTable(xml, shared) {
  const rows = [...String(xml).matchAll(/<row\b[^>]*>([\s\S]*?)<\/row>/g)];
  return rows.map((row) => [...row[1].matchAll(/<c\b([^>]*)>([\s\S]*?)<\/c>/g)]
    .map((cell) => cellValue(cell[1], cell[2], shared))
    .join('\t')
    .trimEnd()).filter((line) => line.replace(/\t/g, '').trim()).join('\n');
}

function cellValue(attrs, body, shared) {
  const type = (/\bt="([^"]+)"/.exec(attrs) || [])[1] || '';
  if (type === 'inlineStr') {
    return [...body.matchAll(/<t\b[^>]*>([\s\S]*?)<\/t>/g)].map((part) => decodeXml(part[1])).join(' ');
  }
  const raw = (/<v\b[^>]*>([\s\S]*?)<\/v>/.exec(body) || [])[1] || '';
  if (type === 's') return shared[Number(raw)] || '';
  if (type === 'b') return raw === '1' ? 'TRUE' : 'FALSE';
  return decodeXml(raw);
}

function zipEntries(buffer) {
  const central = centralEntries(buffer);
  if (central.length) return central;
  return localEntries(buffer);
}

function centralEntries(buffer) {
  const eocd = buffer.lastIndexOf(Buffer.from([0x50, 0x4b, 0x05, 0x06]));
  if (eocd < 0 || eocd + 22 > buffer.length) return [];
  const count = buffer.readUInt16LE(eocd + 10);
  let offset = buffer.readUInt32LE(eocd + 16);
  const entries = [];
  for (let index = 0; index < count && offset + 46 <= buffer.length; index += 1) {
    if (buffer.readUInt32LE(offset) !== 0x02014b50) break;
    const method = buffer.readUInt16LE(offset + 10);
    const compressed = buffer.readUInt32LE(offset + 20);
    const nameLength = buffer.readUInt16LE(offset + 28);
    const extraLength = buffer.readUInt16LE(offset + 30);
    const commentLength = buffer.readUInt16LE(offset + 32);
    const localOffset = buffer.readUInt32LE(offset + 42);
    const name = buffer.subarray(offset + 46, offset + 46 + nameLength).toString('utf8');
    entries.push({ name, text: inflateLocal(buffer, localOffset, method, compressed) });
    offset += 46 + nameLength + extraLength + commentLength;
  }
  return entries;
}

function localEntries(buffer) {
  const entries = [];
  let offset = 0;
  while (offset + 30 < buffer.length && buffer.readUInt32LE(offset) === 0x04034b50) {
    const method = buffer.readUInt16LE(offset + 8);
    const compressed = buffer.readUInt32LE(offset + 18);
    const nameLength = buffer.readUInt16LE(offset + 26);
    const extraLength = buffer.readUInt16LE(offset + 28);
    const nameStart = offset + 30;
    const dataStart = nameStart + nameLength + extraLength;
    const name = buffer.subarray(nameStart, nameStart + nameLength).toString('utf8');
    const data = buffer.subarray(dataStart, Math.min(buffer.length, dataStart + compressed));
    entries.push({ name, text: inflate(method, data) });
    offset = dataStart + compressed;
  }
  return entries;
}

function inflateLocal(buffer, offset, method, compressed) {
  if (offset + 30 > buffer.length || buffer.readUInt32LE(offset) !== 0x04034b50) return '';
  const nameLength = buffer.readUInt16LE(offset + 26);
  const extraLength = buffer.readUInt16LE(offset + 28);
  const dataStart = offset + 30 + nameLength + extraLength;
  return inflate(method, buffer.subarray(dataStart, Math.min(buffer.length, dataStart + compressed)));
}

function inflate(method, data) {
  if (method === 0) return data.toString('utf8');
  if (method !== 8) return '';
  try { return inflateRawSync(data).toString('utf8'); } catch { return ''; }
}

function xmlText(xml) {
  return decodeXml(String(xml || '').replace(/<[^>]+>/g, ' ')).replace(/[ \t]{2,}/g, ' ').trim();
}

function decodeXml(value) {
  return String(value || '')
    .replace(/&#x([0-9a-f]+);/gi, (_, hex) => safeChar(parseInt(hex, 16)))
    .replace(/&#(\d+);/g, (_, num) => safeChar(Number(num)))
    .replace(/&amp;/g, '&')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&apos;/g, "'");
}

function safeChar(code) {
  if (!Number.isFinite(code) || code < 0 || code > 0x10ffff) return '';
  return String.fromCodePoint(code);
}

function binaryStrings(buffer) {
  const found = [];
  let utf = '';
  for (let index = 0; index + 1 < buffer.length; index += 2) {
    const code = buffer.readUInt16LE(index);
    if ((code >= 32 && code <= 126) || (code >= 160 && code <= 591)) utf += String.fromCharCode(code);
    else {
      if (utf.length >= 4) found.push(utf);
      utf = '';
    }
  }
  if (utf.length >= 4) found.push(utf);
  return [...new Set(found)].join('\n').trim();
}

export function secretsMatch(provided, expected) {
  const left = String(provided || '');
  const right = String(expected || '');
  if (!left || !right || left.length !== right.length) return false;
  let mismatch = 0;
  for (let i = 0; i < right.length; i += 1) {
    mismatch |= left.charCodeAt(i) ^ right.charCodeAt(i);
  }
  return mismatch === 0;
}

export function sourceKind(name) {
  const ext = extension(name);
  if (['png', 'jpg', 'jpeg', 'gif', 'webp', 'heic'].includes(ext)) return 'Image';
  if (['mp4', 'webm', 'mov', 'mpeg'].includes(ext)) return 'Video';
  if (['mp3', 'wav', 'm4a'].includes(ext)) return 'Audio';
  if (ext === 'pdf') return 'PDF';
  if (['doc', 'docx'].includes(ext)) return 'Word';
  if (['xls', 'xlsx', 'csv', 'tsv'].includes(ext)) return 'Sheet';
  if (['ppt', 'pptx', 'odp'].includes(ext)) return 'Slides';
  if (['odt'].includes(ext)) return 'Word';
  return 'File';
}

export function saveUploads(root, files) {
  const dir = path.join(root, 'output', 'uploads');
  mkdirSync(dir, { recursive: true });
  const saved = [];
  for (const file of files || []) {
    const name = cleanName(file?.name);
    const bytes = decode(file?.data);
    if (!name || !bytes) {
      saved.push({ ok: false, name: name || 'file', error: 'That file had no readable bytes.' });
      continue;
    }
    if (bytes.length > MAX_BYTES) {
      saved.push({ ok: false, name, error: `${name} is over 18 MB. It was not added.` });
      continue;
    }
    const id = randomUUID();
    const meta = { id, name, mime: mimeFor(name, file?.mime), ext: extension(name) };
    writeFileSync(path.join(dir, `${id}.bin`), bytes);
    writeFileSync(path.join(dir, `${id}.json`), JSON.stringify(meta));
    saved.push({ ok: true, id, name, kind: sourceKind(name) });
  }
  return saved;
}

export function deleteUpload(root, id) {
  if (!/^[0-9a-f-]{36}$/i.test(String(id || ''))) return false;
  const dir = path.join(root, 'output', 'uploads');
  let removed = false;
  for (const suffix of ['.bin', '.json']) {
    try {
      unlinkSync(path.join(dir, `${id}${suffix}`));
      removed = true;
    } catch {
      // The pair is already gone.
    }
  }
  return removed;
}

export function loadUpload(root, id) {
  if (!/^[0-9a-f-]{36}$/i.test(String(id || ''))) return null;
  const dir = path.join(root, 'output', 'uploads');
  try {
    const meta = JSON.parse(readFileSync(path.join(dir, `${id}.json`), 'utf8'));
    const bytes = readFileSync(path.join(dir, `${id}.bin`));
    return { ...meta, bytes };
  } catch {
    return null;
  }
}

export function uploadPart(record) {
  const ext = record.ext || extension(record.name);
  if (TEXT.has(ext) || ['csv', 'tsv'].includes(ext)) {
    return { text: `Source file ${record.name}:\n${record.bytes.toString('utf8').slice(0, 80000)}` };
  }
  if (OFFICE.has(ext)) {
    const text = officeText(record.bytes);
    return { text: `Source file ${record.name}:\n${text || 'No readable text was extracted from this office file.'}`, tabular: ['csv', 'xls', 'xlsx', 'tsv', 'ods'].includes(ext) };
  }
  if (INLINE.has(ext)) {
    return {
      inline: true,
      mime: record.mime || MIME[ext] || 'application/octet-stream',
      data: record.bytes.toString('base64'),
      tabular: false,
    };
  }
  const asText = record.bytes.toString('utf8');
  if (!asText.includes('\u0000')) return { text: `Source file ${record.name}:\n${asText.slice(0, 80000)}` };
  return { text: `Source file ${record.name} is attached as ${record.mime || ext || 'a binary file'}. Use the filename only. Do not invent its contents.` };
}

function extension(name) {
  const match = String(name || '').toLowerCase().match(/\.([a-z0-9]{1,8})$/);
  return match ? match[1] : '';
}

function cleanName(name) {
  return String(name || '').replace(/[\u0000-\u001f]/g, '').trim().slice(0, 180);
}

function mimeFor(name, mime) {
  const given = String(mime || '').trim().toLowerCase();
  if (given && given !== 'application/octet-stream') return given.slice(0, 120);
  return MIME[extension(name)] || 'application/octet-stream';
}

function decode(data) {
  const raw = String(data || '');
  if (!raw) return null;
  try {
    return Buffer.from(raw, 'base64');
  } catch {
    return null;
  }
}

export async function postAppsScript(url, secret, payload, timeoutMs = 120000) {
  let current = url;
  const body = JSON.stringify(payload);
  for (let hop = 0; hop < 5; hop += 1) {
    const response = await fetch(current, {
      method: 'POST',
      redirect: 'manual',
      headers: {
        'Content-Type': 'application/json',
        'X-Webhook-Secret': secret,
      },
      body,
      signal: AbortSignal.timeout(timeoutMs),
    });
    if (response.status >= 300 && response.status < 400) {
      const location = response.headers.get('location');
      if (!location) {
        return { status: 502, body: { ok: false, error: 'Apps Script redirected without a new address.' } };
      }
      current = new URL(location, current).href;
      continue;
    }
    const raw = await response.text();
    let parsed;
    try {
      parsed = JSON.parse(raw);
    } catch {
      return { status: 502, body: { ok: false, error: 'Apps Script returned a non-JSON response.' } };
    }
    if (!parsed.ok) return { status: parsed.error === 'Unauthorized webhook.' ? 401 : 409, body: parsed };
    return { status: 200, body: parsed };
  }
  return { status: 502, body: { ok: false, error: 'Apps Script redirected too many times.' } };
}

===== README.md =====
# Lead generation skeleton

Local Node.js starts a Gemini Deep Research run, polls until the report is ready, and POSTs the markdown to a Google Apps Script web app. The script files leads into Sheets, writes a briefing doc for high-priority accounts, drafts an outreach email, geocodes headquarters, attaches the briefing to matching calendar events, and creates a follow-up task.

```
.env
  |
  v
index.js
  DRY_RUN / REPORT_FILE / RESUME -> existing Node paths
  otherwise                      -> python -m fallback
                                    BILLING_ENABLED and probe ok -> node index.js (LEADGEN_ROUTE=paid)
                                      interactions.create + poll
                                    DEMO_MODE -> fallback/demo_leads.json
                                    otherwise -> Flash planner, Search grounding, reviewer, markdown
  save output/last-report.md
  POST JSON { secret, interactionId, markdown, seller }
  |
  v
Apps Script doPost
  authorize secret
  parse the markdown table
  Sheets  Docs  Drive  Gmail draft  Maps  Calendar  Tasks
```

`gemini-2.5-flash` grounding on the free tier is 500 requests per day, shared with Flash-Lite. The 5,000-search monthly figure on Google's pricing page applies to Gemini 3 and newer models. The fallback records grounded calls in `output/grounding-quota.json` and stops at the published cap for `FREE_MODEL`.

## Free-tier boundary

Sheets, Docs, Drive, Gmail drafts, the built-in Maps geocoder, Calendar, and Tasks run inside a normal Google account and Apps Script quotas. No Maps Platform key and no paid Workspace add-on is required.

The Gemini side is narrower than "free":

- The Interactions API has a free tier. On that tier, stored interactions are retained for one day. Background Deep Research requires `store: true`, so a failed webhook must be resumed or replayed inside that window.
- Google's Deep Research documentation prices a typical `deep-research-preview-04-2026` task on a pay-as-you-go basis (their published estimate is about $1–$3, and the Max agent is higher). This skeleton does not treat that call as free. Confirm the key's project in AI Studio before `DRY_RUN=false`.
- Visualizations are off, collaborative planning is off, and the Max agent is not the default. `DRY_RUN=true` exercises the webhook without calling Gemini. `RESUME_INTERACTION_ID` and `REPORT_FILE` recover a run that already finished.

Apps Script consumer executions are limited to about six minutes. The webhook processes at most 15 leads, scans at most 500 calendar events, and updates at most five of them.

## Security

The web app has to be deployed as "Anyone" so Node can POST without a Google user session. `WEBHOOK_SECRET` is checked before any write. Use a random string of at least 16 characters, put the same value in `.env` and in Apps Script script properties, and treat the `/exec` URL like a credential.

The research prompt tells the agent to leave unknown emails as `unknown`. The script does not invent a recipient. When no public email exists, the draft is addressed to the script owner and prefixed `[Needs recipient]`. Nothing is sent.

Sheet cells that start with `=`, `+`, `-`, or `@` are prefixed so a model-written table cannot become a formula.

## Local setup

Requires Node.js 22 or newer (required by `@google/genai` 3.x).

```powershell
copy .env.example .env
npm install
npm start
```

With the example `.env`, `DRY_RUN=true` and the webhook URL is still a placeholder, so the first start fails until you either clear `GOOGLE_WEBHOOK_URL` (local fixture only) or paste a real `/exec` URL and secret.

A live run needs `DRY_RUN=false`, `GEMINI_API_KEY`, `SELLER_OFFERING`, `TARGET_INDUSTRY`, the webhook URL, and the secret.

If the process is interrupted after the interaction id is printed, the research continues on Google's side:

```powershell
# set RESUME_INTERACTION_ID in .env, then
npm start
```

If the report was saved and the webhook failed:

```powershell
# REPORT_FILE=output/last-report.md
npm start
```

Mode order is `REPORT_FILE`, then `DRY_RUN`, then `RESUME_INTERACTION_ID`, then a new research task.

## Apps Script setup

1. Create a project at [script.google.com](https://script.google.com).
2. Replace `Code.gs` with `apps-script/Code.gs`.
3. Open Project Settings, show the manifest, and replace it with `apps-script/appsscript.json`. Set `timeZone` to your timezone. `Etc/GMT-3` is UTC+3. Task due dates use this zone.
4. In the editor, open Services and add the Google Tasks API if the manifest did not enable it. The symbol must be `Tasks`.
5. Run `install` and approve the scopes. The log prints the spreadsheet URL and the "Leads Briefings" folder URL.
6. In Project Settings > Script properties, set `WEBHOOK_SECRET` to the same value as `.env`. Optional: `HOME_BASE_ADDRESS` (used for distance), `SELLER_NAME`, `SELLER_OFFERING`, `CALENDAR_LOOKAHEAD_DAYS`, `MAX_LEADS`, `TASKLIST_ID`.
7. Deploy > New deployment > Web app. Execute as **Me**. Who has access: **Anyone**. Copy the URL that ends in `/exec`.
8. Open that URL in a browser. The JSON says whether the secret, spreadsheet, and folder are configured. It does not reveal the secret.

`install` creates the spreadsheet and folder on first run and stores their ids in script properties. Run it before the first POST so the OAuth consent happens in the editor. A POST that arrives before consent gets an HTML error page; Node reports that as a non-JSON webhook response and leaves `output/last-report.md` in place.

## Lead contract

The agent is asked for one markdown table and a `## Briefing: {Company}` section for each High row. Structured output is not available on this agent, so the table is the contract.

| Company | Domain | HQ Address | Contact Name | Contact Title | Contact Email | Pain Point | Hook | Priority | Source |

Priority `High` creates the briefing doc, files it in Drive, scans the primary calendar, and creates "Review pitch deck for [Company]". Other rows are still written to Sheets. A draft is created only when the hook cell has public text.

Google Tasks stores a due **date**, not a time. The task is due on the calendar date of the instant 48 hours after processing, in the script timezone. The exact UTC instant is written in the task notes.

A repeated delivery of the same `interactionId` returns `duplicate: true` and does not create a second set of rows. Briefing files, drafts, and tasks are also looked up by stable title before a second one is created.

## Failure behavior

| What happened | What to do |
| --- | --- |
| Process killed while polling | `RESUME_INTERACTION_ID` |
| Webhook rejected the payload | Fix Apps Script, then `REPORT_FILE=output/last-report.md` |
| Report had no table | The file is saved and the webhook is not called |
| One lead's geocode, doc, draft, or task failed | That row is still written. The Notes column and the JSON `errors` array carry the message |
| Script lock was busy | Response is `retryable: true`. Node retries the POST with backoff |

function send(response, status, payload, type = 'application/json; charset=utf-8') {

function webhookConfig() {
  const url = (process.env.GOOGLE_WEBHOOK_URL || '').trim();
  const secret = (process.env.WEBHOOK_SECRET || '').trim();
  return { url, secret };
}

function missingWebhook(action) {
  const { url, secret } = webhookConfig();
  const tail = action === 'save' ? 'Nothing was saved in Gmail.' : 'Nothing was read from Google.';
  if (!url && !secret) return `GOOGLE_WEBHOOK_URL and WEBHOOK_SECRET are empty. ${tail}`;
  if (!url) return `GOOGLE_WEBHOOK_URL is empty. ${tail}`;
  if (!secret) return `WEBHOOK_SECRET is empty. Paste the same string you saved in Apps Script. ${tail}`;
  return '';
}

async function callAppsScript(payload) {
  const { url, secret } = webhookConfig();
  const missing = missingWebhook('read');
  if (missing) return { status: 424, body: { ok: false, error: missing } };
  try {
    return await postAppsScript(url, secret, payload);
  } catch {
    return { status: 502, body: { ok: false, error: 'The desk could not reach Apps Script.' } };
  }
}

async function postApprove(company, interactionId, seller = {}) {
  const { url, secret } = webhookConfig();
  const missing = missingWebhook('save');
  if (missing) {
    return { status: 424, body: { ok: false, synced: false, sent: false, error: missing } };
  }
  try {
    const outcome = await postAppsScript(url, secret, {
      action: 'approve',
      company,
      interactionId,
      ...(seller.name || seller.offering ? { seller } : {}),
    });
    if (!outcome.body?.ok) return { status: outcome.status, body: { ...outcome.body, sent: false, synced: false } };
    return { status: 200, body: { ...outcome.body, sent: false, synced: true } };
  } catch {
    return { status: 502, body: { ok: false, sent: false, error: 'The approve call did not reach Apps Script. Nothing was sent.' } };
  }
}

async function handleApi(request, response, url) {
  if (request.method === 'GET' && url.pathname === '/api/health') {
    send(response, 200, { ok: true, service: 'lead-desk' });
    return;
  }
  if (request.method === 'GET' && url.pathname === '/api/leads') {
    send(response, 200, { ok: true, leads: allLeads(root) });
    return;
  }
  const runMatch = url.pathname.match(/^\/api\/runs\/([^/]+)$/);
  if (request.method === 'GET' && runMatch) {
    const record = runs.get(decodeURIComponent(runMatch[1]));
    if (!record) {
      send(response, 404, { ok: false, error: 'That run is not on this desk.' });
      return;
    }
    send(response, 200, record);
    return;
  }
  const headerToken = request.headers['x-desk-token'];
  const presented = (Array.isArray(headerToken) ? headerToken[0] : headerToken) || cookieToken(request);
  const production = process.env.RENDER === 'true';
  if (!deskAuthorized(presented, expectedDeskToken(), production)) {
    send(response, 401, { ok: false, error: 'Desk token was rejected.' });
    return;
  }
  const metered = url.pathname === '/api/runs' || url.pathname === '/api/ask' || /\/approve$/.test(url.pathname);
  if (metered && limited()) {
    send(response, 429, { ok: false, error: 'Too many desk writes. Wait a few minutes. Nothing was sent.' });
    return;
  }
  if (request.method === 'POST' && url.pathname === '/api/sources') {
    let body = {};
    try {
      body = await readJson(request);
    } catch {
      send(response, 400, { ok: false, error: 'The file body was not JSON. Nothing was added.' });
      return;
    }
    const files = Array.isArray(body.files) ? body.files.slice(0, 12) : [];
    const sources = saveUploads(root, files);
    send(response, sources.some((item) => item.ok) ? 200 : 422, { ok: sources.some((item) => item.ok), sources });
    return;
  }
  const sourceMatch = url.pathname.match(/^\/api\/sources\/([^/]+)$/);
  if (request.method === 'DELETE' && sourceMatch) {
    const removed = deleteUpload(root, decodeURIComponent(sourceMatch[1]));
    send(response, removed ? 200 : 404, removed
      ? { ok: true }
      : { ok: false, error: 'That source is not on this desk.' });
    return;
  }
  if (request.method === 'POST' && url.pathname === '/api/google') {
    let body = {};
    try {
      body = await readJson(request);
    } catch {
      send(response, 400, { ok: false, error: 'The Google request was not JSON.' });
      return;
    }
    const action = body.action === 'list' ? 'list' : 'links';
    const payload = action === 'list'
      ? { action, kinds: Array.isArray(body.kinds) ? body.kinds.slice(0, 5) : [] }
      : { action: 'links' };
    const outcome = await callAppsScript(payload);
    send(response, outcome.status, outcome.body);
    return;
  }
  if (request.method === 'POST' && url.pathname === '/api/ask') {
    let body = {};
    try {
      body = await readJson(request);
    } catch {
      send(response, 400, { ok: false, error: 'The question was not JSON. Nothing was asked.' });
      return;
    }
    const question = String(body.question || '').replace(/[\u0000-\u001f]/g, ' ').trim().slice(0, 2000);
    if (!question) {
      send(response, 422, { ok: false, error: 'The question was empty. Nothing was asked.' });
      return;
    }
    const apiKey = (process.env.GEMINI_API_KEY || '').trim();
    if (!apiKey || /^(your-|paste-)/i.test(apiKey)) {
      send(response, 424, { ok: false, error: 'GEMINI_API_KEY is empty. Nothing was asked.' });
      return;
    }
    const uploads = (Array.isArray(body.uploads) ? body.uploads : []).slice(0, 8).map((id) => loadUpload(root, id)).filter(Boolean);
    const workspace = (Array.isArray(body.workspace) ? body.workspace : []).slice(0, 6);
    let texts = [];
    if (workspace.length) {
      const read = await callAppsScript({ action: 'read', ids: workspace });
      if (!read.body?.ok) {
        send(response, read.status, read.body);
        return;
      }
      texts = Array.isArray(read.body.texts) ? read.body.texts : [];
    }
    try {
      const built = partsForAsk(uploads, texts);
      const answer = await askSources({
        apiKey,
        model: (process.env.FREE_MODEL || 'gemini-2.5-flash').trim() || 'gemini-2.5-flash',
        question,
        profile: businessEnv(body),
        parts: built.parts,
        urls: built.urls,
        tabular: built.tabular,
      });
      const leads = rememberLeads(root, answer.leads);
      send(response, 200, { ok: true, explanation: answer.explanation, leads });
    } catch (error) {
      const message = redactSecrets(error instanceof Error ? error.message : '', [apiKey]).slice(0, 300);
      send(response, 502, { ok: false, error: message || 'Gemini did not answer. Nothing was written to Sheets.' });
    }
    return;
  }
  if (request.method === 'POST' && url.pathname === '/api/runs') {
    let body = {};
    try {
      body = await readJson(request);
    } catch {
      send(response, 400, { ok: false, synced: false, error: 'Run body was not JSON. Nothing was written to Sheets.' });
      return;
    }
    const id = randomUUID();
    const outputDir = path.join(root, 'output', 'runs', id);
    const run = await executeRun(outputDir, { ...process.env, ...businessEnv(body) });
    const stored = rememberLeads(root, run.leads || []);
    const result = run.result || {
      ok: false,
      synced: false,
      error: 'The batch did not report a webhook result. Nothing was written to Sheets.',
    };
    const record = {
      id,
      ok: Boolean(result.ok && result.synced !== false),
      synced: Boolean(result.synced),
      interactionId: result.interactionId || stored[0]?.interactionId || '',
      error: result.error || '',
      appsScript: result.appsScript || null,
      leads: stored,
    };
    runs.set(id, record);
    send(response, record.ok ? 200 : 424, record);
    return;
  }
  const approveMatch = url.pathname.match(/^\/api\/leads\/([^/]+)\/approve$/);
  if (request.method === 'POST' && approveMatch) {
    const id = decodeURIComponent(approveMatch[1]);
    const lead = findLead(root, id);
    if (!lead) {
      send(response, 404, { ok: false, sent: false, error: 'That lead is not on this desk.' });
      return;
    }
    let body = {};
    try {
      body = await readJson(request);
    } catch {
      send(response, 400, { ok: false, sent: false, error: 'Approve body was not JSON. Nothing was sent.' });
      return;
    }
    if (lead.negative || !lead.hook) {
      send(response, 422, { ok: false, sent: false, error: 'A negative match does not create a draft. Nothing was sent.' });
      return;
    }
    const interactionId = String(body.interactionId || '').trim();
    if (!interactionId) {
      send(response, 424, { ok: false, sent: false, error: 'Run research and wait for Apps Script before approving. Nothing was saved in Gmail.' });
      return;
    }
    try {
      const profile = businessEnv(body);
      const outcome = await postApprove(lead.company, interactionId, {
        name: profile.SELLER_NAME || '',
        offering: profile.SELLER_OFFERING || '',
      });
      send(response, outcome.status, outcome.body);
    } catch {
      send(response, 502, { ok: false, sent: false, error: 'The approve call did not reach Apps Script. Nothing was sent.' });
    }
    return;
  }
  send(response, 404, { ok: false, error: 'No such desk route.' });
}

function partsForAsk(uploads, texts) {
  const parts = [];
  let tabular = false;
  let urls = false;
  let bytes = 0;
  for (const record of uploads) {
    const part = uploadPart(record);
    if (part.tabular) tabular = true;
    if (part.text) parts.push({ text: part.text });
    else if (part.inline) {
      bytes += record.bytes.length;
      if (bytes > 18 * 1024 * 1024) {
        throw new Error('The checked files together are over 18 MB. Uncheck one and ask again.');
      }
      parts.push({ inlineData: { mimeType: part.mime, data: part.data } });
    }
  }
  for (const item of texts) {
    if (/^https?:\/\//i.test(item.url || '')) urls = true;
    parts.push({ text: `Google source ${item.name || item.id}:\n${item.text || ''}` });
  }
  return { parts, tabular, urls };
}

async function handleStatic(response, pathname) {
  const requested = pathname === '/' ? '/index.html' : pathname;
  const target = path.resolve(webRoot, `.${requested}`);
  if (target !== webRoot && !target.startsWith(webRoot + path.sep)) {
    send(response, 403, { ok: false, error: 'That path is outside the desk.' });
    return;
  }
  try {
    const file = await readFile(target);
    send(response, 200, file, TYPES[path.extname(target).toLowerCase()] || 'application/octet-stream');
  } catch {
    send(response, 404, { ok: false, error: 'That file is not on the desk.' });
  }
}

export function createDeskServer() {
  return createServer(async (request, response) => {
    try {
      const url = new URL(request.url || '/', 'http://127.0.0.1');
      if (url.pathname.startsWith('/api/')) {
        await handleApi(request, response, url);
        return;
      }
      if (request.method !== 'GET' && request.method !== 'HEAD') {
        send(response, 405, { ok: false, error: 'The desk files only accept GET.' });
        return;
      }
      await handleStatic(response, decodeURIComponent(url.pathname));
    } catch (error) {
      send(response, 500, { ok: false, error: 'The desk hit an unexpected error.' });
    }
  });
}

    console.log(`Lead desk listening on ${port}`);
  });
}

  }

  }

    return leads.map((lead) => `<article class="doc-sheet"><p><b>${escapeHtml(lead.company)}</b> ${escapeHtml(lead.domain)}</p><p>${escapeHtml(lead.hq)} · ${escapeHtml(lead.distance)}</p></article>`).join("");

function paintPhases(active, failed) {
  document.getElementById("phases").innerHTML = phases.map(([id, label], index) => {

    let cls = "phase";
    if (failed && id === active) cls += failed === "limit" ? " bad" : " warn";
    else if (index < activeIndex) cls += " done";

function startRun(extra) {
  clearTimeout(timer);
  phase = "plan";
  if (inSheet()) openSheet("run");
  else show("studio");
  const focus = extra && extra.company ? extra.company : "";
  step("plan", focus ? "Researching " + focus : "Posting the research run", "The desk uses the Research settings on the server. A draft is created only when you approve.");
  postDesk("/api/runs", Object.assign(profileBody(), extra || {}))
    .then(async (response) => ({ response, body: await response.json() }))
    .then(({ response, body }) => {
      if (!response.ok || !body.ok || body.synced === false) {
        const message = body.error || "Apps Script did not accept the batch. Nothing was written to Sheets.";
        step("plan", "The batch was not written", message);
        toast(message);
        return;
      }
      batchInteractionId = body.interactionId || "";
      if (Array.isArray(body.leads) && body.leads.length) mergeLeads(body.leads);
      else leads.forEach((lead) => { if (!lead.interactionId) lead.interactionId = batchInteractionId; });
      step("plan", "Apps Script accepted the batch", "High rows have a briefing and a task. A draft is created only when you approve. Nothing was sent.");
      toast("Apps Script accepted the batch. Nothing was sent.");
      reveal("pipeline");
    })
    .catch((error) => {
      step("plan", "The desk could not reach the API", error.message);
      toast("The desk could not reach the API. Nothing was written to Sheets.");
    });
}

function act(name, id) {
  const lead = leads.find((item) => item.id === id);
  if (!lead) return;
  if (name === "report") { selected = lead; show("lead"); return; }
  if (name === "research") {
    const profile = profileBody();
    startRun({
      company: lead.company,
      domain: lead.domain,
      sellerName: profile.sellerName,
      offering: profile.offering,
      icp: profile.icp,
      industry: profile.industry,
      region: profile.region,
      maxLeads: profile.maxLeads
    });
    return;
  }
  if (name === "archive") {
    lead.archived = true;
    toast(lead.company + " archived on this desk. Workspace files were not deleted.");
    renderBoard();
    if (inSheet()) openSheet("pipeline");
    return;
  }
  if (name === "approve") {

    approveLead(lead);
  }
}

function deskHeaders() {
  const headers = { "Content-Type": "application/json" };
  let token = "";
  try { token = localStorage.getItem("lead-desk-token") || ""; } catch (error) { token = ""; }
  if (!token && window.deskToken) token = String(window.deskToken);
  if (token) headers["X-Desk-Token"] = token;
  return headers;
}
function deskSend(method, url, body, allowPrompt) {
  return fetch(url, {
    method: method,
    headers: deskHeaders(),
    body: body == null ? undefined : JSON.stringify(body)
  }).then((response) => {
    if (response.status !== 401 || !allowPrompt) return response;
    const token = window.prompt("Enter the desk token");
    if (!token) return response;
    try { localStorage.setItem("lead-desk-token", token.trim()); } catch (error) { window.deskToken = token.trim(); }
    return deskSend(method, url, body, false);
  });
}
function postDesk(url, body) {
  return deskSend("POST", url, body, true);
}

}

function escapeHtml(value) {
  return String(value).replace(/[&<>"']/g, (ch) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", "\"": "&quot;", "'": "&#39;" }[ch]));
}
function safeUrl(value) {
  try {
    const url = new URL(String(value || ""), "http://127.0.0.1");
    if (url.protocol !== "https:" && url.protocol !== "http:") return "";
    if (!/^https?:/i.test(String(value || ""))) return "";
    return escapeHtml(url.href);
  } catch {
    return "";
  }
}

document.getElementById("sourceFiles").onchange = async () => {
  const input = document.getElementById("sourceFiles");
  const files = Array.from(input.files || []);
  input.value = "";
  if (!files.length) return;
  const tooBig = files.filter((file) => file.size > 18 * 1024 * 1024);
  tooBig.forEach((file) => toast(file.name + " is over 18 MB. It was not added."));
  const accepted = files.filter((file) => file.size <= 18 * 1024 * 1024);
  if (!accepted.length) return;
  try {
    const payload = await Promise.all(accepted.map(fileToPayload));
    const response = await postDesk("/api/sources", { files: payload });
    const body = await response.json();
    if (!response.ok || !body.ok) {
      toast((body.sources && body.sources.find((item) => item.error) || {}).error || body.error || "The file was not added.");
      return;
    }
    (body.sources || []).forEach((item) => {
      if (!item.ok) {
        toast(item.error || "The file was not added.");
        return;
      }
      sources.push({ id: "src-" + (sourceSeq++), name: item.name, kind: item.kind, saved: true, serverId: item.id });
    });
    renderSources();
    toast(accepted.length + (accepted.length === 1 ? " file added." : " files added."));
  } catch (error) {
    toast(error.message || "The file was not added.");
  }
};
function fileToPayload(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => {
      const value = String(reader.result || "");
      resolve({ name: file.name, mime: file.type || "application/octet-stream", data: value.slice(value.indexOf(",") + 1) });
    };
    reader.onerror = () => reject(new Error(file.name + " could not be read."));
    reader.readAsDataURL(file);
  });
}

document.getElementById("askForm").onsubmit = async (event) => {
  event.preventDefault();
  const input = document.getElementById("askInput");
  const text = input.value.trim();
  if (!text) return;
  const box = document.getElementById("docText");
  const asked = document.createElement("p");
  asked.textContent = text;
  input.value = "";
  box.append(asked);
  const profile = profileBody();
  const checked = sources.filter((item) => item.saved);
  try {
    const response = await postDesk("/api/ask", {
      question: text,
      uploads: checked.filter((item) => item.serverId).map((item) => item.serverId),
      workspace: checked.filter((item) => item.workspaceId).map((item) => ({ id: item.workspaceId, name: item.name, url: item.url || "" })),
      sellerName: profile.sellerName,
      offering: profile.offering,
      icp: profile.icp,
      industry: profile.industry,
      region: profile.region,
      maxLeads: profile.maxLeads
    });
    const body = await response.json();
    if (!response.ok || !body.ok) {
      const reply = document.createElement("p");
      reply.textContent = body.error || "Gemini did not answer. Nothing was written to Sheets.";
      box.append(reply);
      toast(reply.textContent);
      return;
    }
    const reply = document.createElement("p");
    reply.textContent = body.explanation || "Gemini returned no explanation.";
    box.append(reply);
    mergeLeads(body.leads || []);
    paintAskedLeads(body.leads || []);
  } catch (error) {
    const reply = document.createElement("p");
    reply.textContent = error.message || "The desk could not reach Gemini.";
    box.append(reply);
  }
  document.querySelector(".doc-main").scrollTop = document.querySelector(".doc-main").scrollHeight;
};
function mergeLeads(rows) {
  (rows || []).forEach((raw) => {
    if (!raw || !raw.company) return;
    const id = raw.id || String(raw.company).toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "") || "lead";
    const lead = {
      id: id,
      company: raw.company,
      domain: raw.domain || "",
      hq: raw.hq || raw.hqAddress || "",
      pain: raw.pain || raw.painPoint || "No public pain point.",
      hook: raw.hook || "",
      priority: raw.priority === "High" ? "High" : "Low",
      source: raw.source || "",
      photo: raw.photo || "",
      negative: raw.negative != null ? raw.negative : !(raw.priority === "High" && raw.hook),
      interactionId: raw.interactionId || batchInteractionId || "",
      task: raw.task || "",
      taskWhy: raw.taskWhy || "",
      distance: raw.distance || "",
      coords: raw.coords || ""
    };
    const index = leads.findIndex((item) => item.id === lead.id);
    if (index >= 0) leads[index] = Object.assign({}, leads[index], lead, { archived: leads[index].archived });
    else leads.push(lead);
  });
}
function paintAskedLeads(rows) {
  const list = document.getElementById("askLeadList");
  if (!rows.length) {
    list.innerHTML = "<p class=\"meta\">No company in the checked sources matched this question.</p>";
    return;
  }
  list.innerHTML = rows.map((lead) => `<article><b>${escapeHtml(lead.company)}</b><span>${escapeHtml(lead.priority)} · ${escapeHtml(lead.pain || "No pain point in the sources.")}</span><span>${escapeHtml(lead.hook || "No hook. A draft is not created.")}</span></article>`).join("");
}
const googleApps = { map: "maps", report: "briefings", slides: "slides", pipeline: "sheet", run: "sheet", workspace: "sheet", tasks: "tasks", draft: "drafts" };
async function openGoogleApp(sheet) {
  const key = googleApps[sheet] || "sheet";
  try {
    const response = await postDesk("/api/google", { action: "links" });
    const body = await response.json();
    const href = body.links && body.links[key];
    if (!response.ok || !body.ok || !href) {
      toast(body.error || "Google did not return a link. Nothing was opened.");
      return;
    }
    window.open(href, "_blank", "noopener");
  } catch {
    toast("The desk could not reach Google. Nothing was opened.");
  }
}
function workspaceCard() {
  const kinds = ["gmail", "docs", "sheets", "drive", "slides"];
  const labels = { gmail: "Gmail", docs: "Docs", sheets: "Sheets", drive: "Drive", slides: "Slides" };
  return `<div class="ws-toggles" id="wsToggles">${kinds.map((kind) => `<label><input type="checkbox" data-kind="${kind}" checked> ${labels[kind]}</label>`).join("")}</div><div id="wsItems"><p>Loading the connected Google account.</p></div>`;
}
async function loadWorkspaceItems() {
  const box = document.getElementById("wsItems");
  const toggles = document.getElementById("wsToggles");
  if (!box || !toggles) return;
  const kinds = Array.from(toggles.querySelectorAll("input:checked")).map((input) => input.dataset.kind);
  toggles.onchange = () => loadWorkspaceItems();
  if (!kinds.length) {
    box.innerHTML = "<p>Turn on Gmail, Docs, Sheets, Drive, or Slides.</p>";
    return;
  }
  box.innerHTML = "<p>Loading the connected Google account.</p>";
  try {
    const response = await postDesk("/api/google", { action: "list", kinds });
    const body = await response.json();
    if (!response.ok || !body.ok) {
      box.innerHTML = `<p>${escapeHtml(body.error || "Google did not return files.")}</p>`;
      return;
    }
    const items = Array.isArray(body.items) ? body.items : [];
    box.innerHTML = items.map((item) => `<div class="ws-row"><span><b>${escapeHtml(item.name)}</b><small>${escapeHtml(item.kind)}</small></span>${item.id ? `<button type="button" data-add-ws="${escapeHtml(item.id)}" data-ws-name="${escapeHtml(item.name)}" data-ws-kind="${escapeHtml(item.kind)}" data-ws-url="${escapeHtml(item.url || "")}">Add</button>` : ""}</div>`).join("") || "<p>Nothing in the checked Google apps.</p>";
    box.querySelectorAll("[data-add-ws]").forEach((button) => {
      button.onclick = () => {
        if (sources.some((item) => item.workspaceId === button.dataset.addWs)) {
          toast("Already in sources.");
          return;
        }
        sources.push({
          id: "src-" + (sourceSeq++),
          name: button.dataset.wsName,
          kind: button.dataset.wsKind,
          saved: true,
          workspaceId: button.dataset.addWs,
          url: button.dataset.wsUrl
        });
        renderSources();
        toast(button.dataset.wsName + " added as a source.");
      };
    });
  } catch {
    box.innerHTML = "<p>The desk could not reach Google.</p>";
  }
}

async function loadDesk() {
  const response = await fetch("/api/leads");
  if (!response.ok) throw new Error("The lead list was refused.");
  const body = await response.json();
  leads = Array.isArray(body.leads) ? body.leads : [];
  selected = leads[0] || null;
  const requested = new URLSearchParams(location.search).get("view");
  if (requested && document.getElementById("view-" + requested)) {
    gate.hidden = true;
    document.getElementById("desk").hidden = false;
    show(requested);
    return;
  }
  if (!document.getElementById("view-studio").hidden) paintView("studio");
}
